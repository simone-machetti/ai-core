\begin{center}
{\LARGE \textbf{Radix-4}}\\[4pt]
{\large DP8 (8$\times$4) --- one digit per pair of bits of $B$}
\end{center}

\vspace{1.5em}

\noindent\textbf{Step 1 --- the dot product}
\[
\sum_{i=0}^{7} A_i \, B_i
\]

\vspace{0.5em}

\noindent\textbf{Step 2 --- group the bits in pairs and exchange the sums}

\noindent Each $B_i$ is a 4-bit value read as signed; each pair of bits is one digit and
the top pair holds the negative bit, $B_i = d_i^{\,0} + 4\,d_i^{\,1}$ with
\[
d_i^{\,0} = 2\,B_i^{\,1} + B_i^{\,0} \in \{0, 1, 2, 3\}
\qquad\qquad
d_i^{\,1} = -\,2\,B_i^{\,3} + B_i^{\,2} \in \{-2, -1, 0, 1\}
\]
\[
\sum_{i=0}^{7} A_i \, B_i
  \;=\; \sum_{i=0}^{7} \Biggl[\, \sum_{j=0}^{1} 4^{j} d_i^{\,j} \Biggr] A_i
  \;=\; \sum_{j=0}^{1} 4^{j} \sum_{i=0}^{7} d_i^{\,j} \, A_i
\]

\noindent When $B_i$ is a lower slice of a wider $B$ the top pair is read unsigned,
$d_i^{\,1} \in \{0, 1, 2, 3\}$.

\vspace{0.5em}

\noindent\textbf{Step 3 --- the partial products}
\[
\sum_{i=0}^{7} A_i \, B_i
  \;=\; \sum_{i=0}^{7}
        \underbrace{\Bigl[\, d_i^{\,0} \, A_i \Bigr]}_{\textstyle \Rl}
  \;+\; 4 \sum_{i=0}^{7}
        \underbrace{\Bigl[\, d_i^{\,1} \, A_i \Bigr]}_{\textstyle \Rh}
\]

\noindent Each takes one of four values; the digit $3$ is the cost, $3A_i = 2A_i + A_i$
needs an adder:

\vspace{0.5em}

\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{c c | c | l}
$B_i^{\,1}$ & $B_i^{\,0}$ & $d_i^{\,0}$ & \multicolumn{1}{c}{$R_0$} \\
\hline
$0$ & $0$ & $0$ & $0$ \\
$0$ & $1$ & $1$ & $A_i$ \\
$1$ & $0$ & $2$ & $2A_i$ \\
$1$ & $1$ & $3$ & $3A_i$ \\
\end{tabular}
\qquad\qquad
\begin{tabular}{c c | c | l}
$B_i^{\,3}$ & $B_i^{\,2}$ & $d_i^{\,1}$ & \multicolumn{1}{c}{$R_1$} \\
\hline
$0$ & $0$ & $0$  & $0$ \\
$0$ & $1$ & $+1$ & $A_i$ \\
$1$ & $0$ & $-2$ & $-2A_i$ \\
$1$ & $1$ & $-1$ & $-A_i$ \\
\end{tabular}
\end{center}

\vspace{0.8em}

\noindent\textbf{Step 4 --- split the work}

\noindent There is nothing to encode: the digit is the bit pair itself, and the dispatcher
sends it to the PE as the control signals of the selection; only the selection stays inside:
\[
\underbrace{\bigl\{\, B_i^{\,1},\, B_i^{\,0} \,\bigr\} \;\longrightarrow\; d_i^{\,0}
  \qquad
  \bigl\{\, B_i^{\,3},\, B_i^{\,2} \,\bigr\} \;\longrightarrow\; d_i^{\,1}}_{\textstyle \text{dispatcher: encoding}}
\qquad\longrightarrow\qquad
\underbrace{\sum_{i=0}^{7} d_i^{\,0} \, A_i \;+\; 4 \sum_{i=0}^{7} d_i^{\,1} \, A_i}_{\textstyle \text{PE: selection}}
\]

\noindent Each $d_i^{\,j}$ travels as its bit pair, which the cell turns into one of four
multiples: 4 control bits per lane. The multiple $3A_i$ is either an adder in every cell or
a third value precomputed once per row, as Bit-Plane does with its pair sums.

\vspace{0.8em}

\noindent $B$ is the narrow operand: 2 digits $\times$ 8 lanes = 16 selections of
10 bits per DP8. Booth and NR4SD are two ways of removing the digit $3$.
