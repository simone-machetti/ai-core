\newcommand{\Rf}{R\bigl(A_{2i}^{\,j},\, A_{2i+1}^{\,j}\bigr)}

\begin{document}

\begin{center}
{\LARGE \textbf{Bit-Plane A}}\\[4pt]
{\large DP8 (8$\times$4) --- decomposition over the bit planes of $A$}
\end{center}

\vspace{1.5em}

\noindent\textbf{Step 1 --- the dot product}
\[
\sum_{i=0}^{7} A_i \, B_i
\]

\vspace{0.5em}

\noindent\textbf{Step 2 --- expand $A$ into its bit planes and exchange the sums}

\noindent Each $A_i$ is an 8-bit value, $A_i = \sum_{j=0}^{7} 2^{j} A_i^{\,j}$, so
\[
\sum_{i=0}^{7} A_i \, B_i
  \;=\; \sum_{i=0}^{7} \Biggl[\, \sum_{j=0}^{7} 2^{j} A_i^{\,j} \Biggr] B_i
  \;=\; \sum_{j=0}^{7} 2^{j} \sum_{i=0}^{7} A_i^{\,j} \, B_i
\]

\noindent $A_i^{\,j} B_i$ is a selection, not a product.

\vspace{0.5em}

\noindent\textbf{Step 3 --- pair the lanes}
\[
\sum_{i=0}^{7} A_i \, B_i
  \;=\; \sum_{j=0}^{7} 2^{j} \sum_{i=0}^{3}
        \underbrace{\Bigl[\, A_{2i}^{\,j} \, B_{2i} \;+\; A_{2i+1}^{\,j} \, B_{2i+1} \Bigr]}_{\textstyle \Rf}
\]

\noindent $\Rf$ depends on the two bits $A_{2i}^{\,j}$, $A_{2i+1}^{\,j}$ and takes one of
four values:

\vspace{1em}

\begin{center}
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{c c | l}
$A_{2i}^{\,j}$ & $A_{2i+1}^{\,j}$ & \multicolumn{1}{c}{$\Rf$} \\
\hline
$0$ & $0$ & $0$ \\
$0$ & $1$ & $B_{2i+1}$ \\
$1$ & $0$ & $B_{2i}$ \\
$1$ & $1$ & $B_{2i} + B_{2i+1}$ \\
\end{tabular}
\end{center}

\vspace{1.5em}

\noindent\textbf{Step 4 --- split the work}

\noindent Every entry of the table is a function of $B$ alone, so the table is built
outside the PE and only the selection stays inside:
\[
\underbrace{\bigl\{\, 0,\; B_{2i},\; B_{2i+1},\; B_{2i} + B_{2i+1} \,\bigr\}}_{\textstyle \text{dispatcher}}
\qquad\longrightarrow\qquad
\underbrace{\sum_{j=0}^{7} 2^{j} \sum_{i=0}^{3} \Rf}_{\textstyle \text{PE}}
\]

\vspace{1.5em}

\noindent $A$ is the wide operand: 8 planes $\times$ 4 lane pairs = 32 selections of
6 bits per DP8, twice the rows of Bit-Plane B.
