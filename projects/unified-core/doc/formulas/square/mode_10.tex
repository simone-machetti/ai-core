\begin{center}
{\LARGE \textbf{Mode 10 --- Square}}\\[4pt]
{\large C-DP8 (C8$\times$C8) $\rightarrow$ DP16 (8$\times$8) $\rightarrow$ DP16 (8$\times$4) $\rightarrow$ DP8 (8$\times$4) $\rightarrow$ DP8 (4$\times$4) $\rightarrow$ add-and-square}
\end{center}

\vspace{0.1em}

\textbf{Step 1 --- C-DP8 (C8$\times$C8)}
\[
\text{C-DP8}\,(\mathrm{C8}\!\times\!\mathrm{C8}) = D = \sum_{i=0}^{7} a_i \, b_i, \qquad
a_i = a_i^{re} + \mathrm{j}\,a_i^{im}, \quad
b_i = b_i^{re} + \mathrm{j}\,b_i^{im}
\]

\textbf{Step 2 --- DP16 (8$\times$8)}
\[
\mathrm{Re}(D) = \underbrace{\sum_{i=0}^{7} a_i^{re}\, b_i^{re}
  - \sum_{i=0}^{7} a_i^{im}\, b_i^{im}}_{\DPop{16}{8}{8}}
\]
\[
\mathrm{Im}(D) = \underbrace{\sum_{i=0}^{7} a_i^{re}\, b_i^{im}
  + \sum_{i=0}^{7} a_i^{im}\, b_i^{re}}_{\DPop{16}{8}{8}}
\]

\textbf{Step 3 --- DP16 (8$\times$4)}
\[
\mathrm{Re}(D) = 2^{4}\underbrace{\sum_{i=0}^{7} a_i^{re}\, b_i^{re,hi}
  - \sum_{i=0}^{7} a_i^{im}\, b_i^{im,hi}}_{\DPop{16}{8}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i^{re}\, b_i^{re,lo}
  - \sum_{i=0}^{7} a_i^{im}\, b_i^{im,lo}}_{\DPop{16}{8}{4}}
\]
\[
\mathrm{Im}(D) = 2^{4}\underbrace{\sum_{i=0}^{7} a_i^{re}\, b_i^{im,hi}
  + \sum_{i=0}^{7} a_i^{im}\, b_i^{re,hi}}_{\DPop{16}{8}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i^{re}\, b_i^{im,lo}
  + \sum_{i=0}^{7} a_i^{im}\, b_i^{re,lo}}_{\DPop{16}{8}{4}}
\]

\textbf{Step 4 --- DP8 (8$\times$4)}
\begin{align*}
\mathrm{Re}(D) = {}
  & 2^{4}\underbrace{\sum_{i=0}^{7} a_i^{re}\, b_i^{re,hi}}_{\DPop{8}{8}{4}}
  - 2^{4}\underbrace{\sum_{i=0}^{7} a_i^{im}\, b_i^{im,hi}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i^{re}\, b_i^{re,lo}}_{\DPop{8}{8}{4}}
  - \underbrace{\sum_{i=0}^{7} a_i^{im}\, b_i^{im,lo}}_{\DPop{8}{8}{4}}
\end{align*}
\begin{align*}
\mathrm{Im}(D) = {}
  & 2^{4}\underbrace{\sum_{i=0}^{7} a_i^{re}\, b_i^{im,hi}}_{\DPop{8}{8}{4}}
  + 2^{4}\underbrace{\sum_{i=0}^{7} a_i^{im}\, b_i^{re,hi}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i^{re}\, b_i^{im,lo}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i^{im}\, b_i^{re,lo}}_{\DPop{8}{8}{4}}
\end{align*}

\textbf{Step 5 --- DP8 (4$\times$4)}
\begin{align*}
\mathrm{Re}(D) = {}
  & 2^{8}\underbrace{\sum_{i=0}^{7} a_{H,i}^{re}\, b_i^{re,hi}}_{\DPop{8}{4}{4}}
  + 2^{4}\underbrace{\sum_{i=0}^{7} a_{L,i}^{re}\, b_i^{re,hi}}_{\DPop{8}{4}{4}}
  - 2^{8}\underbrace{\sum_{i=0}^{7} a_{H,i}^{im}\, b_i^{im,hi}}_{\DPop{8}{4}{4}}
  - 2^{4}\underbrace{\sum_{i=0}^{7} a_{L,i}^{im}\, b_i^{im,hi}}_{\DPop{8}{4}{4}} \\[0.2em]
  & + 2^{4}\underbrace{\sum_{i=0}^{7} a_{H,i}^{re}\, b_i^{re,lo}}_{\DPop{8}{4}{4}}
  + \underbrace{\sum_{i=0}^{7} a_{L,i}^{re}\, b_i^{re,lo}}_{\DPop{8}{4}{4}}
  - 2^{4}\underbrace{\sum_{i=0}^{7} a_{H,i}^{im}\, b_i^{im,lo}}_{\DPop{8}{4}{4}}
  - \underbrace{\sum_{i=0}^{7} a_{L,i}^{im}\, b_i^{im,lo}}_{\DPop{8}{4}{4}}
\end{align*}
\begin{align*}
\mathrm{Im}(D) = {}
  & 2^{8}\underbrace{\sum_{i=0}^{7} a_{H,i}^{re}\, b_i^{im,hi}}_{\DPop{8}{4}{4}}
  + 2^{4}\underbrace{\sum_{i=0}^{7} a_{L,i}^{re}\, b_i^{im,hi}}_{\DPop{8}{4}{4}}
  + 2^{8}\underbrace{\sum_{i=0}^{7} a_{H,i}^{im}\, b_i^{re,hi}}_{\DPop{8}{4}{4}}
  + 2^{4}\underbrace{\sum_{i=0}^{7} a_{L,i}^{im}\, b_i^{re,hi}}_{\DPop{8}{4}{4}} \\[0.2em]
  & + 2^{4}\underbrace{\sum_{i=0}^{7} a_{H,i}^{re}\, b_i^{im,lo}}_{\DPop{8}{4}{4}}
  + \underbrace{\sum_{i=0}^{7} a_{L,i}^{re}\, b_i^{im,lo}}_{\DPop{8}{4}{4}}
  + 2^{4}\underbrace{\sum_{i=0}^{7} a_{H,i}^{im}\, b_i^{re,lo}}_{\DPop{8}{4}{4}}
  + \underbrace{\sum_{i=0}^{7} a_{L,i}^{im}\, b_i^{re,lo}}_{\DPop{8}{4}{4}}
\end{align*}

\textbf{Step 6 --- Add-and-square}\quad
{\small(substitute each DP8 (4$\times$4) $= \tfrac{1}{2}\big[\sum(x_i{+}b_i)^2 - \sum x_i^2 - \sum b_i^2\big]$)}
\begin{align*}
\mathrm{Re}(D) = {}
  & 2^{8}\underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{H,i}^{re}{+}b_i^{re,hi})^2 - \sum_{i=0}^{7}(a_{H,i}^{re})^2 - \sum_{i=0}^{7}(b_i^{re,hi})^2\Big)}_{\DPop{8}{4}{4}} \\[0.2em]
  & + 2^{4}\underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{L,i}^{re}{+}b_i^{re,hi})^2 - \sum_{i=0}^{7}(a_{L,i}^{re})^2 - \sum_{i=0}^{7}(b_i^{re,hi})^2\Big)}_{\DPop{8}{4}{4}} \\[0.2em]
  & + 2^{8}\underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{H,i}^{im}{-}b_i^{im,hi})^2 - \sum_{i=0}^{7}(a_{H,i}^{im})^2 - \sum_{i=0}^{7}(b_i^{im,hi})^2\Big)}_{\DPop{8}{4}{4}} \\[0.2em]
  & + 2^{4}\underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{L,i}^{im}{-}b_i^{im,hi})^2 - \sum_{i=0}^{7}(a_{L,i}^{im})^2 - \sum_{i=0}^{7}(b_i^{im,hi})^2\Big)}_{\DPop{8}{4}{4}} \\[0.2em]
  & + 2^{4}\underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{H,i}^{re}{+}b_i^{re,lo})^2 - \sum_{i=0}^{7}(a_{H,i}^{re})^2 - \sum_{i=0}^{7}(b_i^{re,lo})^2\Big)}_{\DPop{8}{4}{4}} \\[0.2em]
  & + \underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{L,i}^{re}{+}b_i^{re,lo})^2 - \sum_{i=0}^{7}(a_{L,i}^{re})^2 - \sum_{i=0}^{7}(b_i^{re,lo})^2\Big)}_{\DPop{8}{4}{4}} \\[0.2em]
  & + 2^{4}\underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{H,i}^{im}{-}b_i^{im,lo})^2 - \sum_{i=0}^{7}(a_{H,i}^{im})^2 - \sum_{i=0}^{7}(b_i^{im,lo})^2\Big)}_{\DPop{8}{4}{4}} \\[0.2em]
  & + \underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{L,i}^{im}{-}b_i^{im,lo})^2 - \sum_{i=0}^{7}(a_{L,i}^{im})^2 - \sum_{i=0}^{7}(b_i^{im,lo})^2\Big)}_{\DPop{8}{4}{4}}
\end{align*}
\begin{align*}
\mathrm{Im}(D) = {}
  & 2^{8}\underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{H,i}^{re}{+}b_i^{im,hi})^2 - \sum_{i=0}^{7}(a_{H,i}^{re})^2 - \sum_{i=0}^{7}(b_i^{im,hi})^2\Big)}_{\DPop{8}{4}{4}} \\[0.2em]
  & + 2^{4}\underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{L,i}^{re}{+}b_i^{im,hi})^2 - \sum_{i=0}^{7}(a_{L,i}^{re})^2 - \sum_{i=0}^{7}(b_i^{im,hi})^2\Big)}_{\DPop{8}{4}{4}} \\[0.2em]
  & + 2^{8}\underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{H,i}^{im}{+}b_i^{re,hi})^2 - \sum_{i=0}^{7}(a_{H,i}^{im})^2 - \sum_{i=0}^{7}(b_i^{re,hi})^2\Big)}_{\DPop{8}{4}{4}} \\[0.2em]
  & + 2^{4}\underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{L,i}^{im}{+}b_i^{re,hi})^2 - \sum_{i=0}^{7}(a_{L,i}^{im})^2 - \sum_{i=0}^{7}(b_i^{re,hi})^2\Big)}_{\DPop{8}{4}{4}} \\[0.2em]
  & + 2^{4}\underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{H,i}^{re}{+}b_i^{im,lo})^2 - \sum_{i=0}^{7}(a_{H,i}^{re})^2 - \sum_{i=0}^{7}(b_i^{im,lo})^2\Big)}_{\DPop{8}{4}{4}} \\[0.2em]
  & + \underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{L,i}^{re}{+}b_i^{im,lo})^2 - \sum_{i=0}^{7}(a_{L,i}^{re})^2 - \sum_{i=0}^{7}(b_i^{im,lo})^2\Big)}_{\DPop{8}{4}{4}} \\[0.2em]
  & + 2^{4}\underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{H,i}^{im}{+}b_i^{re,lo})^2 - \sum_{i=0}^{7}(a_{H,i}^{im})^2 - \sum_{i=0}^{7}(b_i^{re,lo})^2\Big)}_{\DPop{8}{4}{4}} \\[0.2em]
  & + \underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{L,i}^{im}{+}b_i^{re,lo})^2 - \sum_{i=0}^{7}(a_{L,i}^{im})^2 - \sum_{i=0}^{7}(b_i^{re,lo})^2\Big)}_{\DPop{8}{4}{4}}
\end{align*}

\textbf{Step 7 --- All components}

\textbf{Re}(D):
\[
\boxed{\;
\begin{aligned}
\mathrm{PE} ={}& 2^{8}\sum_{i=0}^{7}(a_{H,i}^{re}{+}b_i^{re,hi})^2 + 2^{4}\sum_{i=0}^{7}(a_{L,i}^{re}{+}b_i^{re,hi})^2 + 2^{8}\sum_{i=0}^{7}(a_{H,i}^{im}{-}b_i^{im,hi})^2 + 2^{4}\sum_{i=0}^{7}(a_{L,i}^{im}{-}b_i^{im,hi})^2 \\
  & + 2^{4}\sum_{i=0}^{7}(a_{H,i}^{re}{+}b_i^{re,lo})^2 + \sum_{i=0}^{7}(a_{L,i}^{re}{+}b_i^{re,lo})^2 + 2^{4}\sum_{i=0}^{7}(a_{H,i}^{im}{-}b_i^{im,lo})^2 + \sum_{i=0}^{7}(a_{L,i}^{im}{-}b_i^{im,lo})^2 \\[0.2em]
\alpha ={}& 2^{8}\sum_{i=0}^{7}(a_{H,i}^{re})^2 + 2^{4}\sum_{i=0}^{7}(a_{L,i}^{re})^2 + 2^{8}\sum_{i=0}^{7}(a_{H,i}^{im})^2 + 2^{4}\sum_{i=0}^{7}(a_{L,i}^{im})^2 \\
  & + 2^{4}\sum_{i=0}^{7}(a_{H,i}^{re})^2 + \sum_{i=0}^{7}(a_{L,i}^{re})^2 + 2^{4}\sum_{i=0}^{7}(a_{H,i}^{im})^2 + \sum_{i=0}^{7}(a_{L,i}^{im})^2 \\[0.2em]
\beta ={}& 2^{8}\sum_{i=0}^{7}(b_i^{re,hi})^2 + 2^{4}\sum_{i=0}^{7}(b_i^{re,hi})^2 + 2^{8}\sum_{i=0}^{7}(b_i^{im,hi})^2 + 2^{4}\sum_{i=0}^{7}(b_i^{im,hi})^2 \\
  & + 2^{4}\sum_{i=0}^{7}(b_i^{re,lo})^2 + \sum_{i=0}^{7}(b_i^{re,lo})^2 + 2^{4}\sum_{i=0}^{7}(b_i^{im,lo})^2 + \sum_{i=0}^{7}(b_i^{im,lo})^2
\end{aligned}
\;}
\]


\textbf{Im}(D):
\[
\boxed{\;
\begin{aligned}
\mathrm{PE} ={}& 2^{8}\sum_{i=0}^{7}(a_{H,i}^{re}{+}b_i^{im,hi})^2 + 2^{4}\sum_{i=0}^{7}(a_{L,i}^{re}{+}b_i^{im,hi})^2 + 2^{8}\sum_{i=0}^{7}(a_{H,i}^{im}{+}b_i^{re,hi})^2 + 2^{4}\sum_{i=0}^{7}(a_{L,i}^{im}{+}b_i^{re,hi})^2 \\
  & + 2^{4}\sum_{i=0}^{7}(a_{H,i}^{re}{+}b_i^{im,lo})^2 + \sum_{i=0}^{7}(a_{L,i}^{re}{+}b_i^{im,lo})^2 + 2^{4}\sum_{i=0}^{7}(a_{H,i}^{im}{+}b_i^{re,lo})^2 + \sum_{i=0}^{7}(a_{L,i}^{im}{+}b_i^{re,lo})^2 \\[0.2em]
\alpha ={}& 2^{8}\sum_{i=0}^{7}(a_{H,i}^{re})^2 + 2^{4}\sum_{i=0}^{7}(a_{L,i}^{re})^2 + 2^{8}\sum_{i=0}^{7}(a_{H,i}^{im})^2 + 2^{4}\sum_{i=0}^{7}(a_{L,i}^{im})^2 \\
  & + 2^{4}\sum_{i=0}^{7}(a_{H,i}^{re})^2 + \sum_{i=0}^{7}(a_{L,i}^{re})^2 + 2^{4}\sum_{i=0}^{7}(a_{H,i}^{im})^2 + \sum_{i=0}^{7}(a_{L,i}^{im})^2 \\[0.2em]
\beta ={}& 2^{8}\sum_{i=0}^{7}(b_i^{im,hi})^2 + 2^{4}\sum_{i=0}^{7}(b_i^{im,hi})^2 + 2^{8}\sum_{i=0}^{7}(b_i^{re,hi})^2 + 2^{4}\sum_{i=0}^{7}(b_i^{re,hi})^2 \\
  & + 2^{4}\sum_{i=0}^{7}(b_i^{im,lo})^2 + \sum_{i=0}^{7}(b_i^{im,lo})^2 + 2^{4}\sum_{i=0}^{7}(b_i^{re,lo})^2 + \sum_{i=0}^{7}(b_i^{re,lo})^2
\end{aligned}
\;}
\]
