\begin{center}
{\LARGE \textbf{Mode 9 --- Baseline-BFP}}\\[4pt]
{\large DP8 (16$\times$16) $\rightarrow$ DP8 (16$\times$8) $\rightarrow$ DP8 (16$\times$4) $\rightarrow$ DP8 (8$\times$4) $\rightarrow$ block exponents}
\end{center}

\vspace{0.3em}

\textbf{Step 1 --- DP8 (16$\times$16)}
\[ \DPop{8}{16}{16} = \sum_{i=0}^{7} a_i \, b_i \]

\textbf{Step 2 --- DP8 (16$\times$8)}
\[
\DPop{8}{16}{16}
  = 2^{8} \underbrace{\sum_{i=0}^{7} a_i \, b_i^{hi}}_{\DPop{8}{16}{8}}
  \; + \; \underbrace{\sum_{i=0}^{7} a_i \, b_i^{lo}}_{\DPop{8}{16}{8}}
\]

\textbf{Step 3 --- DP8 (16$\times$4)}
\[
\DPop{8}{16}{16}
  = 2^{12} \underbrace{\sum_{i=0}^{7} a_i \, b_i^{hh}}_{\DPop{8}{16}{4}}
  + 2^{8} \underbrace{\sum_{i=0}^{7} a_i \, b_i^{hl}}_{\DPop{8}{16}{4}}
  + 2^{4} \underbrace{\sum_{i=0}^{7} a_i \, b_i^{lh}}_{\DPop{8}{16}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i \, b_i^{ll}}_{\DPop{8}{16}{4}}
\]

\textbf{Step 4 --- DP8 (8$\times$4)}
\begin{align*}
\DPop{8}{16}{16} = {}
  & 2^{20} \underbrace{\sum_{i=0}^{7} a_i^{hi} b_i^{hh}}_{\DPop{8}{8}{4}}
  + 2^{16} \underbrace{\sum_{i=0}^{7} a_i^{hi} b_i^{hl}}_{\DPop{8}{8}{4}}
  + 2^{12} \underbrace{\sum_{i=0}^{7} a_i^{lo} b_i^{hh}}_{\DPop{8}{8}{4}}
  + 2^{12} \underbrace{\sum_{i=0}^{7} a_i^{hi} b_i^{lh}}_{\DPop{8}{8}{4}} \\[0.3em]
  & + 2^{8} \underbrace{\sum_{i=0}^{7} a_i^{lo} b_i^{hl}}_{\DPop{8}{8}{4}}
  + 2^{8} \underbrace{\sum_{i=0}^{7} a_i^{hi} b_i^{ll}}_{\DPop{8}{8}{4}}
  + 2^{4} \underbrace{\sum_{i=0}^{7} a_i^{lo} b_i^{lh}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i^{lo} b_i^{ll}}_{\DPop{8}{8}{4}}
\end{align*}

\textbf{Step 5 --- Block exponents}\quad
{\small(one exponent block per DP8 (8$\times$4) --- all copies of one $E$; parallel matmul: own $E_a,E_b$)}
\[
\big(P_j,\; E\big)\ \ \text{per}\ \DPop{8}{8}{4}\ \text{above}, \qquad E = E_a + E_b
\;\longrightarrow\;
\big(M,\; E\big)
\]
$\delta\equiv0$: $M$ is the Step-4 sum unchanged; \quad $\DPop{8}{16}{16}=M\cdot2^{E}$.

\textbf{Step 6 --- Result}
\[
\boxed{\;
\begin{aligned}
M ={}& 2^{20}\!\sum_{i=0}^{7} a_i^{hi} b_i^{hh}
  + 2^{16}\!\sum_{i=0}^{7} a_i^{hi} b_i^{hl}
  + 2^{12}\!\sum_{i=0}^{7} a_i^{lo} b_i^{hh}
  + 2^{12}\!\sum_{i=0}^{7} a_i^{hi} b_i^{lh} \\
  & + 2^{8}\!\sum_{i=0}^{7} a_i^{lo} b_i^{hl}
  + 2^{8}\!\sum_{i=0}^{7} a_i^{hi} b_i^{ll}
  + 2^{4}\!\sum_{i=0}^{7} a_i^{lo} b_i^{lh}
  + \sum_{i=0}^{7} a_i^{lo} b_i^{ll},
\qquad E = E_a + E_b
\end{aligned}
\;}
\]
