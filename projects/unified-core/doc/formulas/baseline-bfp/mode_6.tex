\begin{center}
{\LARGE \textbf{Mode 6 --- Baseline-BFP}}\\[4pt]
{\large DP32 (8$\times$8) $\rightarrow$ DP16 (8$\times$8) $\rightarrow$ DP16 (8$\times$4) $\rightarrow$ DP8 (8$\times$4) $\rightarrow$ block exponents}
\end{center}

\vspace{0.3em}

\textbf{Step 1 --- DP32 (8$\times$8)}
\[ \DPop{32}{8}{8} = \sum_{i=0}^{31} a_i \, b_i \]

\textbf{Step 2 --- DP16 (8$\times$8)}
\[
\DPop{32}{8}{8}
  = \underbrace{\sum_{i=0}^{15} a_i \, b_i}_{\DPop{16}{8}{8}}
  \; + \; \underbrace{\sum_{i=16}^{31} a_i \, b_i}_{\DPop{16}{8}{8}}
\]

\textbf{Step 3 --- DP16 (8$\times$4)}
\[
\DPop{32}{8}{8}
  = 2^{4} \underbrace{\sum_{i=0}^{15} a_i \, b_i^{hi}}_{\DPop{16}{8}{4}}
  + \underbrace{\sum_{i=0}^{15} a_i \, b_i^{lo}}_{\DPop{16}{8}{4}}
  + 2^{4} \underbrace{\sum_{i=16}^{31} a_i \, b_i^{hi}}_{\DPop{16}{8}{4}}
  + \underbrace{\sum_{i=16}^{31} a_i \, b_i^{lo}}_{\DPop{16}{8}{4}}
\]

\textbf{Step 4 --- DP8 (8$\times$4)}
\begin{align*}
\DPop{32}{8}{8} = {}
  & 2^{4} \underbrace{\sum_{i=0}^{7} a_i \, b_i^{hi}}_{\DPop{8}{8}{4}}
  + 2^{4} \underbrace{\sum_{i=8}^{15} a_i \, b_i^{hi}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i \, b_i^{lo}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=8}^{15} a_i \, b_i^{lo}}_{\DPop{8}{8}{4}} \\[0.3em]
  & + 2^{4} \underbrace{\sum_{i=16}^{23} a_i \, b_i^{hi}}_{\DPop{8}{8}{4}}
  + 2^{4} \underbrace{\sum_{i=24}^{31} a_i \, b_i^{hi}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=16}^{23} a_i \, b_i^{lo}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=24}^{31} a_i \, b_i^{lo}}_{\DPop{8}{8}{4}}
\end{align*}

\textbf{Step 5 --- Block exponents}\quad
{\small(one exponent block per DP8 (8$\times$4))}
\[
\Big(2^{4}\!\!\sum_{i=8q}^{8q+7}\! a_i b_i^{hi} + \!\!\sum_{i=8q}^{8q+7}\! a_i b_i^{lo},\; E_q\Big)_{q=0..3}
\;\longrightarrow\;
\Big(\, M ,\; E \,\Big)
\qquad
E_q = E_{a,q} + E_{b,q}
\]
$E=\max(E_0,\dots,E_3)$, \ $\delta_q=E-E_q$ (staged in the tree, the stage shifts telescope); \quad truncating shifts, on block sums only; \quad $\DPop{32}{8}{8}=M\cdot2^{E}$.

\textbf{Step 6 --- Result}
\[
\boxed{\;
\begin{aligned}
M ={}& 2^{-\delta_0}\Big(2^{4}\!\sum_{i=0}^{7} a_i b_i^{hi} + \sum_{i=0}^{7} a_i b_i^{lo}\Big)
  + 2^{-\delta_1}\Big(2^{4}\!\sum_{i=8}^{15} a_i b_i^{hi} + \sum_{i=8}^{15} a_i b_i^{lo}\Big) \\
  & + 2^{-\delta_2}\Big(2^{4}\!\sum_{i=16}^{23} a_i b_i^{hi} + \sum_{i=16}^{23} a_i b_i^{lo}\Big)
  + 2^{-\delta_3}\Big(2^{4}\!\sum_{i=24}^{31} a_i b_i^{hi} + \sum_{i=24}^{31} a_i b_i^{lo}\Big), \\[0.3em]
E ={}& \max(E_0,E_1,E_2,E_3)
\end{aligned}
\;}
\]
