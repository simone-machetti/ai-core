\begin{center}
{\LARGE \textbf{Mode 1 --- Baseline-BFP}}\\[4pt]
{\large DP16 (8$\times$4) $\rightarrow$ DP8 (8$\times$4) $\rightarrow$ block exponents}
\end{center}

\vspace{0.5em}

\textbf{Step 1 --- DP16 (8$\times$4)}
\[ \DPop{16}{8}{4} = \sum_{i=0}^{15} a_i \, b_i \]

\vspace{0.5em}

\textbf{Step 2 --- DP8 (8$\times$4)}
\[
\DPop{16}{8}{4}
  = \underbrace{\sum_{i=0}^{7} a_i \, b_i}_{\DPop{8}{8}{4}}
  \; + \; \underbrace{\sum_{i=8}^{15} a_i \, b_i}_{\DPop{8}{8}{4}}
\]

\vspace{0.5em}

\textbf{Step 3 --- Block exponents}\quad
{\small(one exponent block per DP8 (8$\times$4))}
\[
\Big(\underbrace{\sum_{i=0}^{7} a_i b_i}_{\DPop{8}{8}{4}},\; E_0\Big) \;,\;
\Big(\underbrace{\sum_{i=8}^{15} a_i b_i}_{\DPop{8}{8}{4}},\; E_1\Big)
\;\longrightarrow\;
\Big(\, M ,\; E \,\Big)
\]
$E_0=E_{a,0}+E_b$, \ $E_1=E_{a,1}+E_b$; \quad $E=\max(E_0,E_1)$, \ $\delta_k=E-E_k$; \quad truncating shifts, on block sums only; \quad $\DPop{16}{8}{4}=M\cdot2^{E}$.

\vspace{0.5em}

\textbf{Step 4 --- Result}
\[
\boxed{\;
M = 2^{-\delta_0}\!\sum_{i=0}^{7} a_i b_i + 2^{-\delta_1}\!\sum_{i=8}^{15} a_i b_i,
\qquad E = \max(E_0,E_1)
\;}
\]
