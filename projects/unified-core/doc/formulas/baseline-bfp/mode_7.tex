\begin{center}
{\LARGE \textbf{Mode 7 --- Baseline-BFP}}\\[4pt]
{\large DP16 (16$\times$8) $\rightarrow$ DP8 (16$\times$8) $\rightarrow$ DP8 (16$\times$4) $\rightarrow$ DP8 (8$\times$4) $\rightarrow$ block exponents}
\end{center}

\vspace{0.3em}

\textbf{Step 1 --- DP16 (16$\times$8)}
\[ \DPop{16}{16}{8} = \sum_{i=0}^{15} a_i \, b_i \]

\textbf{Step 2 --- DP8 (16$\times$8)}
\[
\DPop{16}{16}{8}
  = \underbrace{\sum_{i=0}^{7} a_i \, b_i}_{\DPop{8}{16}{8}}
  \; + \; \underbrace{\sum_{i=8}^{15} a_i \, b_i}_{\DPop{8}{16}{8}}
\]

\textbf{Step 3 --- DP8 (16$\times$4)}
\[
\DPop{16}{16}{8}
  = 2^{4} \underbrace{\sum_{i=0}^{7} a_i \, b_i^{hi}}_{\DPop{8}{16}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i \, b_i^{lo}}_{\DPop{8}{16}{4}}
  + 2^{4} \underbrace{\sum_{i=8}^{15} a_i \, b_i^{hi}}_{\DPop{8}{16}{4}}
  + \underbrace{\sum_{i=8}^{15} a_i \, b_i^{lo}}_{\DPop{8}{16}{4}}
\]

\textbf{Step 4 --- DP8 (8$\times$4)}
\begin{align*}
\DPop{16}{16}{8} = {}
  & 2^{12} \underbrace{\sum_{i=0}^{7} a_i^{hi} \, b_i^{hi}}_{\DPop{8}{8}{4}}
  + 2^{8} \underbrace{\sum_{i=0}^{7} a_i^{hi} \, b_i^{lo}}_{\DPop{8}{8}{4}}
  + 2^{4} \underbrace{\sum_{i=0}^{7} a_i^{lo} \, b_i^{hi}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i^{lo} \, b_i^{lo}}_{\DPop{8}{8}{4}} \\[0.3em]
  & + 2^{12} \underbrace{\sum_{i=8}^{15} a_i^{hi} \, b_i^{hi}}_{\DPop{8}{8}{4}}
  + 2^{8} \underbrace{\sum_{i=8}^{15} a_i^{hi} \, b_i^{lo}}_{\DPop{8}{8}{4}}
  + 2^{4} \underbrace{\sum_{i=8}^{15} a_i^{lo} \, b_i^{hi}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=8}^{15} a_i^{lo} \, b_i^{lo}}_{\DPop{8}{8}{4}}
\end{align*}

\textbf{Step 5 --- Block exponents}\quad
{\small(one exponent block per DP8 (8$\times$4))}
\[
\Big(2^{12}\!\sum_{i=0}^{7} a_i^{hi} b_i^{hi} + 2^{8}\!\sum_{i=0}^{7} a_i^{hi} b_i^{lo} + 2^{4}\!\sum_{i=0}^{7} a_i^{lo} b_i^{hi} + \sum_{i=0}^{7} a_i^{lo} b_i^{lo},\; E_0\Big) ,\;
\Big(\cdots\ i\!\in\![8,15]\ \cdots,\; E_1\Big)
\;\longrightarrow\;
\Big(\, M ,\; E \,\Big)
\]
$E_k=E_{a,k}+E_{b,k}$; \quad $E=\max(E_0,E_1)$, \ $\delta_k=E-E_k$; \quad truncating shifts, on block sums only; \quad $\DPop{16}{16}{8}=M\cdot2^{E}$.

\textbf{Step 6 --- Result}
\[
\boxed{\;
\begin{aligned}
M ={}& 2^{-\delta_0}\Big(2^{12}\!\sum_{i=0}^{7} a_i^{hi} b_i^{hi} + 2^{8}\!\sum_{i=0}^{7} a_i^{hi} b_i^{lo} + 2^{4}\!\sum_{i=0}^{7} a_i^{lo} b_i^{hi} + \sum_{i=0}^{7} a_i^{lo} b_i^{lo}\Big) \\
  & + 2^{-\delta_1}\Big(2^{12}\!\sum_{i=8}^{15} a_i^{hi} b_i^{hi} + 2^{8}\!\sum_{i=8}^{15} a_i^{hi} b_i^{lo} + 2^{4}\!\sum_{i=8}^{15} a_i^{lo} b_i^{hi} + \sum_{i=8}^{15} a_i^{lo} b_i^{lo}\Big), \\[0.3em]
E ={}& \max(E_0,E_1)
\end{aligned}
\;}
\]
