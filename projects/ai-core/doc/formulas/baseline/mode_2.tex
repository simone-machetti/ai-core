\newcommand{\DPop}[3]{\mathrm{DP#1}\,(#2\!\times\!#3)}

\begin{document}

\begin{center}
{\LARGE \textbf{Mode 2}}\\[4pt]
{\large DP16 (8$\times$8) $\rightarrow$ DP16 (8$\times$4) $\rightarrow$ DP8 (8$\times$4)}
\end{center}

\vspace{4em}

\noindent\textbf{Step 1 --- DP16 (8$\times$8)}
\[
\DPop{16}{8}{8} = \sum_{i=0}^{15} a_i \, b_i
\]

\vspace{0.5em}

\noindent\textbf{Step 2 --- DP16 (8$\times$4)}
\[
\DPop{16}{8}{8}
  = 2^{4} \underbrace{\sum_{i=0}^{15} a_i \, b_i^{hi}}_{\DPop{16}{8}{4}}
  \; + \; \underbrace{\sum_{i=0}^{15} a_i \, b_i^{lo}}_{\DPop{16}{8}{4}}
\]

\vspace{0.5em}

\noindent\textbf{Step 3 --- DP8 (8$\times$4)}
\[
\DPop{16}{8}{8}
  = 2^{4} \underbrace{\sum_{i=0}^{7} a_i \, b_i^{hi}}_{\DPop{8}{8}{4}}
  + 2^{4} \underbrace{\sum_{i=8}^{15} a_i \, b_i^{hi}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i \, b_i^{lo}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=8}^{15} a_i \, b_i^{lo}}_{\DPop{8}{8}{4}}
\]
