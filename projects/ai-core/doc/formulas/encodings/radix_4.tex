\begin{center}
{\LARGE \textbf{Radix 4}}\\[4pt]
{\large plain binary --- one digit per pair of bits of $B$}
\end{center}

\vspace{3em}

\noindent\textbf{Step 1 --- the number}

\noindent $B$ is an $n$-bit two's complement value ($n$ even). The top bit counts negative:
\[
B \;=\; -\,2^{n-1} B^{\,n-1} \;+\; \sum_{j=0}^{n-2} 2^{j} \, B^{\,j}
\]

\vspace{0.5em}

\noindent\textbf{Step 2 --- group the bits in pairs}

\noindent Each pair of bits is one digit. The top pair holds the negative bit:
\[
B \;=\; \sum_{j=0}^{n/2-1} 4^{j} \, d_j
\qquad
\begin{aligned}
d_j       &= 2 \, B^{\,2j+1} + B^{\,2j} \;\in\; \{0, 1, 2, 3\}       & j &< n/2 - 1 \\
d_{n/2-1} &= -\,2 \, B^{\,n-1} + B^{\,n-2} \;\in\; \{-2, -1, 0, 1\}
\end{aligned}
\]

\vspace{0.5em}

\noindent\textbf{Step 3 --- the product}
\[
A \, B \;=\; \sum_{j=0}^{n/2-1} 4^{j} \, d_j A
\]

\noindent $d_j A$ takes one of four values in each position:

\vspace{1em}

\begin{center}
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{c c | r l | r l}
& & \multicolumn{2}{c|}{lower digits} & \multicolumn{2}{c}{top digit} \\
$B^{\,2j+1}$ & $B^{\,2j}$ & $d_j$ & $d_j A$ & $d_{n/2-1}$ & $d_{n/2-1} A$ \\
\hline
$0$ & $0$ & $0$ & $0$  & $0$  & $0$ \\
$0$ & $1$ & $1$ & $A$  & $1$  & $A$ \\
$1$ & $0$ & $2$ & $2A$ & $-2$ & $-2A$ \\
$1$ & $1$ & $3$ & $3A$ & $-1$ & $-A$ \\
\end{tabular}
\end{center}

\vspace{1.5em}

\noindent $n/2$ digits, $n/2$ partial products. The digit $3$ is the problem:
$3A = 2A + A$ needs an adder per partial product. Booth and NR4SD are two ways
of removing it.
