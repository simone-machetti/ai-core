\begin{center}
{\LARGE \textbf{NR4SD$^{+}$ / Booth, 2 PPs}}\\[4pt]
{\large DP8 (8$\times$4) --- NR4SD$^{+}$ low digit, Booth top digit, two per nibble}
\end{center}

\vspace{1.5em}

\noindent\textbf{Step 1 --- the dot product}
\[
\sum_{i=0}^{7} A_i \, B_i
\]

\vspace{0.5em}

\noindent\textbf{Step 2 --- recode $B$ into radix-4 digits and exchange the sums}

\noindent Each $B_i$ is a 4-bit value read as signed, plus a carry in $c_i$. The low pair is
recoded NR4SD$^{+}$ and pushes a carry $c_i'$ into the top pair, which is read as a Booth
digit:
\[
s_i = 2\,B_i^{\,1} + B_i^{\,0} + c_i
\qquad
c_i' = \bigl[\, s_i \geq 3 \,\bigr]
\qquad
d_i^{\,0} = s_i - 4\,c_i' \in \{-1, 0, 1, 2\}
\qquad
d_i^{\,1} = -\,2\,B_i^{\,3} + B_i^{\,2} + c_i' \in \{-2,\dots,2\}
\]
\[
\sum_{i=0}^{7} A_i \, B_i
  \;=\; \sum_{i=0}^{7} \Biggl[\, \sum_{j=0}^{1} 4^{j} d_i^{\,j} \Biggr] A_i
  \;=\; \sum_{j=0}^{1} 4^{j} \sum_{i=0}^{7} d_i^{\,j} \, A_i
\]

\noindent Two NR4SD$^{+}$ digits span $[-5, 10]$ and miss $-6, -7, -8$; the Booth top digit
brings the $-2$, so two digits span $[-9, 10]$ and cover a signed nibble plus carry in.
The carry in $c_i$ is $0$ for a whole 4-bit operand and the top bit of the nibble below
when $B_i$ is a slice of a wider $B$ (Step 5).

\vspace{0.5em}

\noindent\textbf{Step 3 --- the partial products}
\[
\sum_{i=0}^{7} A_i \, B_i
  \;=\; \sum_{i=0}^{7}
        \underbrace{\Bigl[\, d_i^{\,0} \, A_i \Bigr]}_{\textstyle \Rl}
  \;+\; 4 \sum_{i=0}^{7}
        \underbrace{\Bigl[\, d_i^{\,1} \, A_i \Bigr]}_{\textstyle \Rh}
\]

\noindent $\Rl$ depends on the low pair and the carry in and takes one of four values;
$\Rh$ is the Booth table with $c_i'$ as the bit below:

\vspace{0.5em}

\vspace{0.5em}

\noindent\textbf{Step 4 --- split the work}

\noindent The encoding is a function of $B$ alone, so it is done in the dispatcher, which
sends the digits $d_i^{\,j}$ --- the control signals of the selection multiplexers --- to the
PE; only the selection of the multiple stays inside:
\[
\underbrace{\bigl\{\, B_i^{\,1},\, B_i^{\,0},\, c_i \,\bigr\} \;\longrightarrow\; d_i^{\,0}
  \qquad
  \bigl\{\, B_i^{\,3},\, B_i^{\,2},\, c_i' \,\bigr\} \;\longrightarrow\; d_i^{\,1}}_{\textstyle \text{dispatcher: encoding}}
\qquad\longrightarrow\qquad
\underbrace{\sum_{i=0}^{7} d_i^{\,0} \, A_i \;+\; 4 \sum_{i=0}^{7} d_i^{\,1} \, A_i}_{\textstyle \text{PE: selection}}
\]

\noindent $d_i^{\,0}$ travels as its 2-bit code
$\bigl(B_i^{\,0} \oplus c_i,\; B_i^{\,1} \oplus (B_i^{\,0} \wedge c_i)\bigr)$, with
$00 \to 0$, $01 \to +1$, $10 \to +2$, $11 \to -1$, which the low cell uses directly;
$d_i^{\,1}$ travels as the Booth window $\{B_i^{\,3}, B_i^{\,2}, c_i'\}$ with
$c_i' = B_i^{\,1} \wedge (B_i^{\,0} \vee c_i)$, which the top cell decodes: 5 control bits per lane.

\vspace{0.5em}

\noindent\textbf{Step 5 --- slices of a wider $B$}

\noindent A lower nibble $L$ read as signed is short by $16\,b_3$; the carry in hands $+b_3$ to the
upper nibble $H$ in the DP8 above, and the $\times 16$ of the tree cancels the two,
element by element, whatever the grouping of the DP8s:
\[
16\,(H + b_3)\,A \;+\; (L - 16\,b_3)\,A \;=\; (16\,H + L)\,A
\]

\vspace{0.5em}

\noindent $B$ is the narrow operand: 2 digits $\times$ 8 lanes = 16 selections of
10 bits per DP8, the low one over $\{0, A_i, 2A_i, -A_i\}$ (2 control bits), the top one
over $\{0, \pm A_i, \pm 2A_i\}$ (3 control bits).
