\begin{center}
{\LARGE \textbf{Mode 3 --- Baseline-BFP}}\\[4pt]
{\large DP8 (16$\times$8) $\rightarrow$ DP8 (16$\times$4) $\rightarrow$ DP8 (8$\times$4) $\rightarrow$ block exponents}
\end{center}

\vspace{0.5em}

\textbf{Step 1 --- DP8 (16$\times$8)}
\[ \DPop{8}{16}{8} = \sum_{i=0}^{7} a_i \, b_i \]

\vspace{0.5em}

\textbf{Step 2 --- DP8 (16$\times$4)}
\[
\DPop{8}{16}{8}
  = 2^{4} \underbrace{\sum_{i=0}^{7} a_i \, b_i^{hi}}_{\DPop{8}{16}{4}}
  \; + \; \underbrace{\sum_{i=0}^{7} a_i \, b_i^{lo}}_{\DPop{8}{16}{4}}
\]

\vspace{0.5em}

\textbf{Step 3 --- DP8 (8$\times$4)}
\[
\DPop{8}{16}{8}
  = 2^{12} \underbrace{\sum_{i=0}^{7} a_i^{hi} \, b_i^{hi}}_{\DPop{8}{8}{4}}
  + 2^{8} \underbrace{\sum_{i=0}^{7} a_i^{hi} \, b_i^{lo}}_{\DPop{8}{8}{4}}
  + 2^{4} \underbrace{\sum_{i=0}^{7} a_i^{lo} \, b_i^{hi}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i^{lo} \, b_i^{lo}}_{\DPop{8}{8}{4}}
\]

\vspace{0.5em}

\textbf{Step 4 --- Block exponents}\quad
{\small(one exponent block per DP8 (8$\times$4) --- all copies of one $E$)}
\[
\big(P_j,\; E\big)\ \ \text{per}\ \DPop{8}{8}{4}\ \text{above}, \qquad E = E_a + E_b
\;\longrightarrow\;
\big(M,\; E\big)
\]
$\delta\equiv0$: $M$ is the Step-3 sum unchanged; \quad $\DPop{8}{16}{8}=M\cdot2^{E}$.

\vspace{0.5em}

\textbf{Step 5 --- Result}
\[
\boxed{\;
M = 2^{12}\!\sum_{i=0}^{7} a_i^{hi} b_i^{hi}
  + 2^{8}\!\sum_{i=0}^{7} a_i^{hi} b_i^{lo}
  + 2^{4}\!\sum_{i=0}^{7} a_i^{lo} b_i^{hi}
  + \sum_{i=0}^{7} a_i^{lo} b_i^{lo},
\qquad E = E_a + E_b
\;}
\]
