\begin{center}
{\LARGE \textbf{Mode 3 --- Square (centered / opt)}}\\[4pt]
{\large DP8 (16$\times$8) $\rightarrow$ DP8 (16$\times$4) $\rightarrow$ DP8 (8$\times$4) $\rightarrow$ DP8 (4$\times$4) $\rightarrow$ centered add-and-square}
\end{center}

\textbf{Step 1 --- DP8 (16$\times$8)}
\[ \DPop{8}{16}{8} = \sum_{i=0}^{7} a_i \, b_i \]

\textbf{Step 2 --- DP8 (16$\times$4)}
\[
\DPop{8}{16}{8}
  = 2^{4}\underbrace{\sum_{i=0}^{7} a_i \, b_i^{hi}}_{\DPop{8}{16}{4}}
  \; + \; \underbrace{\sum_{i=0}^{7} a_i \, b_i^{lo}}_{\DPop{8}{16}{4}}
\]

\textbf{Step 3 --- DP8 (8$\times$4)}
\begin{align*}
\DPop{8}{16}{8} = {}
  & 2^{12}\underbrace{\sum_{i=0}^{7} a_i^{hi} \, b_i^{hi}}_{\DPop{8}{8}{4}}
  + 2^{8}\underbrace{\sum_{i=0}^{7} a_i^{hi} \, b_i^{lo}}_{\DPop{8}{8}{4}}
  + 2^{4}\underbrace{\sum_{i=0}^{7} a_i^{lo} \, b_i^{hi}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i^{lo} \, b_i^{lo}}_{\DPop{8}{8}{4}}
\end{align*}

\textbf{Step 4 --- DP8 (4$\times$4)}
\begin{align*}
\DPop{8}{16}{8} = {}
  & 2^{16}\underbrace{\sum_{i=0}^{7} a_{H,i}^{hi}\, b_i^{hi}}_{\DPop{8}{4}{4}}
  + 2^{12}\underbrace{\sum_{i=0}^{7} a_{L,i}^{hi}\, b_i^{hi}}_{\DPop{8}{4}{4}}
  + 2^{12}\underbrace{\sum_{i=0}^{7} a_{H,i}^{hi}\, b_i^{lo}}_{\DPop{8}{4}{4}}
  + 2^{8}\underbrace{\sum_{i=0}^{7} a_{L,i}^{hi}\, b_i^{lo}}_{\DPop{8}{4}{4}} \\[0.6em]
  & + 2^{8}\underbrace{\sum_{i=0}^{7} a_{H,i}^{lo}\, b_i^{hi}}_{\DPop{8}{4}{4}}
  + 2^{4}\underbrace{\sum_{i=0}^{7} a_{L,i}^{lo}\, b_i^{hi}}_{\DPop{8}{4}{4}}
  + 2^{4}\underbrace{\sum_{i=0}^{7} a_{H,i}^{lo}\, b_i^{lo}}_{\DPop{8}{4}{4}}
  + \underbrace{\sum_{i=0}^{7} a_{L,i}^{lo}\, b_i^{lo}}_{\DPop{8}{4}{4}}
\end{align*}

\textbf{Step 5 --- Centered add-and-square}\quad
{\small(subtract $8$ from unsigned nibbles $a_L^{hi},a_H^{lo},a_L^{lo},b^{lo}$; per DP8 (4$\times$4) add $c=0/512/2048$ for 0/1/2 unsigned nibbles)}
\begin{align*}
\DPop{8}{16}{8} = {}
  & 2^{16}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}(a_{H,i}^{hi}{+}b_i^{hi})^2 - \sum_{i=0}^{7}(a_{H,i}^{hi})^2 - \sum_{i=0}^{7}(b_i^{hi})^2\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + 2^{12}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big((a_{L,i}^{hi}{-}8){+}b_i^{hi}\big)^2 - \sum_{i=0}^{7}(a_{L,i}^{hi}{-}8)^2 - \sum_{i=0}^{7}(b_i^{hi}{-}8)^2 + 512\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + 2^{12}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big(a_{H,i}^{hi}{+}(b_i^{lo}{-}8)\big)^2 - \sum_{i=0}^{7}(a_{H,i}^{hi}{-}8)^2 - \sum_{i=0}^{7}(b_i^{lo}{-}8)^2 + 512\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + 2^{8}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big((a_{L,i}^{hi}{-}8){+}(b_i^{lo}{-}8)\big)^2 - \sum_{i=0}^{7}(a_{L,i}^{hi}{-}16)^2 - \sum_{i=0}^{7}(b_i^{lo}{-}16)^2 + 2048\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + 2^{8}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big((a_{H,i}^{lo}{-}8){+}b_i^{hi}\big)^2 - \sum_{i=0}^{7}(a_{H,i}^{lo}{-}8)^2 - \sum_{i=0}^{7}(b_i^{hi}{-}8)^2 + 512\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + 2^{4}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big((a_{L,i}^{lo}{-}8){+}b_i^{hi}\big)^2 - \sum_{i=0}^{7}(a_{L,i}^{lo}{-}8)^2 - \sum_{i=0}^{7}(b_i^{hi}{-}8)^2 + 512\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + 2^{4}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big((a_{H,i}^{lo}{-}8){+}(b_i^{lo}{-}8)\big)^2 - \sum_{i=0}^{7}(a_{H,i}^{lo}{-}16)^2 - \sum_{i=0}^{7}(b_i^{lo}{-}16)^2 + 2048\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + \underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big((a_{L,i}^{lo}{-}8){+}(b_i^{lo}{-}8)\big)^2 - \sum_{i=0}^{7}(a_{L,i}^{lo}{-}16)^2 - \sum_{i=0}^{7}(b_i^{lo}{-}16)^2 + 2048\Big)}_{\DPop{8}{4}{4}}
\end{align*}

\textbf{Step 6 --- All components}
\[
\boxed{\;
\begin{aligned}
\mathrm{PE} ={}& 2^{16}\!\sum_{i=0}^{7}(a_{H,i}^{hi}{+}b_i^{hi})^2 + 2^{12}\!\sum_{i=0}^{7}\big((a_{L,i}^{hi}{-}8){+}b_i^{hi}\big)^2 + 2^{12}\!\sum_{i=0}^{7}\big(a_{H,i}^{hi}{+}(b_i^{lo}{-}8)\big)^2 + 2^{8}\!\sum_{i=0}^{7}\big((a_{L,i}^{hi}{-}8){+}(b_i^{lo}{-}8)\big)^2 \\
  & + 2^{8}\!\sum_{i=0}^{7}\big((a_{H,i}^{lo}{-}8){+}b_i^{hi}\big)^2 + 2^{4}\!\sum_{i=0}^{7}\big((a_{L,i}^{lo}{-}8){+}b_i^{hi}\big)^2 + 2^{4}\!\sum_{i=0}^{7}\big((a_{H,i}^{lo}{-}8){+}(b_i^{lo}{-}8)\big)^2 + \sum_{i=0}^{7}\big((a_{L,i}^{lo}{-}8){+}(b_i^{lo}{-}8)\big)^2 \\[0.5em]
\alpha ={}& 2^{16}\!\sum_{i=0}^{7}(a_{H,i}^{hi})^2 + 2^{12}\!\sum_{i=0}^{7}(a_{L,i}^{hi}{-}8)^2 + 2^{12}\!\sum_{i=0}^{7}(a_{H,i}^{hi}{-}8)^2 + 2^{8}\!\sum_{i=0}^{7}(a_{L,i}^{hi}{-}16)^2 \\
  & + 2^{8}\!\sum_{i=0}^{7}(a_{H,i}^{lo}{-}8)^2 + 2^{4}\!\sum_{i=0}^{7}(a_{L,i}^{lo}{-}8)^2 + 2^{4}\!\sum_{i=0}^{7}(a_{H,i}^{lo}{-}16)^2 + \sum_{i=0}^{7}(a_{L,i}^{lo}{-}16)^2 \\[0.5em]
\beta ={}& 2^{16}\!\sum_{i=0}^{7}(b_i^{hi})^2 + 2^{12}\!\sum_{i=0}^{7}(b_i^{hi}{-}8)^2 + 2^{12}\!\sum_{i=0}^{7}(b_i^{lo}{-}8)^2 + 2^{8}\!\sum_{i=0}^{7}(b_i^{lo}{-}16)^2 \\
  & + 2^{8}\!\sum_{i=0}^{7}(b_i^{hi}{-}8)^2 + 2^{4}\!\sum_{i=0}^{7}(b_i^{hi}{-}8)^2 + 2^{4}\!\sum_{i=0}^{7}(b_i^{lo}{-}16)^2 + \sum_{i=0}^{7}(b_i^{lo}{-}16)^2 \\[0.5em]
C ={}& 0 + 2097152 + 2097152 + 524288 + 131072 + 8192 + 32768 + 2048 = 4892672
\end{aligned}
\;}
\]
