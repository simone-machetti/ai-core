{\LARGE \textbf{Mode 11 --- Square (centered / opt)}}\\[4pt]
{\large C-DP16 (C8$\times$C8) $\rightarrow$ DP32 (8$\times$8) $\rightarrow$ DP16 (8$\times$8) $\rightarrow$ DP16 (8$\times$4) $\rightarrow$ DP8 (8$\times$4) $\rightarrow$ DP8 (4$\times$4) $\rightarrow$ centered add-and-square}

\textbf{Step 1 --- C-DP16 (C8$\times$C8)}
\[
\text{C-DP16}\,(\mathrm{C8}\!\times\!\mathrm{C8}) = D = \sum_{i=0}^{15} a_i \, b_i, \qquad
a_i = a_i^{re} + \mathrm{j}\,a_i^{im}, \quad
b_i = b_i^{re} + \mathrm{j}\,b_i^{im}
\]

\textbf{Step 2 --- DP32 (8$\times$8)}
\[
\mathrm{Re}(D) = \underbrace{\sum_{i=0}^{15} a_i^{re} b_i^{re}
  - \sum_{i=0}^{15} a_i^{im} b_i^{im}}_{\DPop{32}{8}{8}}
\]
\[
\mathrm{Im}(D) = \underbrace{\sum_{i=0}^{15} a_i^{re} b_i^{im}
  + \sum_{i=0}^{15} a_i^{im} b_i^{re}}_{\DPop{32}{8}{8}}
\]

\textbf{Step 3 --- DP16 (8$\times$8)}
\[
\mathrm{Re}(D) = \underbrace{\sum_{i=0}^{15} a_i^{re} b_i^{re}}_{\DPop{16}{8}{8}}
  \; - \; \underbrace{\sum_{i=0}^{15} a_i^{im} b_i^{im}}_{\DPop{16}{8}{8}}
\]
\[
\mathrm{Im}(D) = \underbrace{\sum_{i=0}^{15} a_i^{re} b_i^{im}}_{\DPop{16}{8}{8}}
  \; + \; \underbrace{\sum_{i=0}^{15} a_i^{im} b_i^{re}}_{\DPop{16}{8}{8}}
\]

\textbf{Step 4 --- DP16 (8$\times$4)}
\begin{align*}
