{\LARGE \textbf{Mode 8 --- Square (centered / opt)}}\\[4pt]
{\large DP16 (16$\times$16) $\rightarrow$ DP8 (16$\times$16) $\rightarrow$ DP8 (16$\times$8) $\rightarrow$ DP8 (16$\times$4) $\rightarrow$ DP8 (8$\times$4) $\rightarrow$ DP8 (4$\times$4) $\rightarrow$ centered add-and-square}
\end{center}

\vspace{0.1em}

\textbf{Step 1 --- DP16 (16$\times$16)}
\[
\DPop{16}{16}{16} = \sum_{i=0}^{15} a_i \, b_i
\]

\textbf{Step 2 --- DP8 (16$\times$16)}
\[
\DPop{16}{16}{16}
  = \underbrace{\sum_{i=0}^{7} a_i \, b_i}_{\DPop{8}{16}{16}}
  \; + \; \underbrace{\sum_{i=8}^{15} a_i \, b_i}_{\DPop{8}{16}{16}}
\]

\textbf{Step 3 --- DP8 (16$\times$8)}
\[
\DPop{16}{16}{16}
  = 2^{8} \underbrace{\sum_{i=0}^{7} a_i \, b_i^{hi}}_{\DPop{8}{16}{8}}
  + \underbrace{\sum_{i=0}^{7} a_i \, b_i^{lo}}_{\DPop{8}{16}{8}}
  + 2^{8} \underbrace{\sum_{i=8}^{15} a_i \, b_i^{hi}}_{\DPop{8}{16}{8}}
  + \underbrace{\sum_{i=8}^{15} a_i \, b_i^{lo}}_{\DPop{8}{16}{8}}
\]
