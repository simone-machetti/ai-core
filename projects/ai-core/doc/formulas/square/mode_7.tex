\begin{center}
{\LARGE \textbf{Mode 7 --- Square}}\\[4pt]
{\large DP16 (16$\times$8) $\rightarrow$ DP8 (16$\times$8) $\rightarrow$ DP8 (16$\times$4) $\rightarrow$ DP8 (8$\times$4) $\rightarrow$ DP8 (4$\times$4) $\rightarrow$ add-and-square}
\end{center}

\vspace{0.1em}

\textbf{Step 1 --- DP16 (16$\times$8)}
\[
\DPop{16}{16}{8} = \sum_{i=0}^{15} a_i \, b_i
\]

\textbf{Step 2 --- DP8 (16$\times$8)}
\[
\DPop{16}{16}{8}
  = \underbrace{\sum_{i=0}^{7} a_i \, b_i}_{\DPop{8}{16}{8}}
  \; + \; \underbrace{\sum_{i=8}^{15} a_i \, b_i}_{\DPop{8}{16}{8}}
\]

\textbf{Step 3 --- DP8 (16$\times$4)}
\begin{align*}
\DPop{16}{16}{8} = {}
  & 2^{4}\underbrace{\sum_{i=0}^{7} a_i \, b_i^{hi}}_{\DPop{8}{16}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i \, b_i^{lo}}_{\DPop{8}{16}{4}}
  + 2^{4}\underbrace{\sum_{i=8}^{15} a_i \, b_i^{hi}}_{\DPop{8}{16}{4}}
  + \underbrace{\sum_{i=8}^{15} a_i \, b_i^{lo}}_{\DPop{8}{16}{4}}
\end{align*}

\textbf{Step 4 --- DP8 (8$\times$4)}
\begin{align*}
\DPop{16}{16}{8} = {}
  & 2^{12}\underbrace{\sum_{i=0}^{7} a_i^{hi} \, b_i^{hi}}_{\DPop{8}{8}{4}}
  + 2^{8}\underbrace{\sum_{i=0}^{7} a_i^{hi} \, b_i^{lo}}_{\DPop{8}{8}{4}}
  + 2^{4}\underbrace{\sum_{i=0}^{7} a_i^{lo} \, b_i^{hi}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i^{lo} \, b_i^{lo}}_{\DPop{8}{8}{4}} \\[0.6em]
  & + 2^{12}\underbrace{\sum_{i=8}^{15} a_i^{hi} \, b_i^{hi}}_{\DPop{8}{8}{4}}
  + 2^{8}\underbrace{\sum_{i=8}^{15} a_i^{hi} \, b_i^{lo}}_{\DPop{8}{8}{4}}
  + 2^{4}\underbrace{\sum_{i=8}^{15} a_i^{lo} \, b_i^{hi}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=8}^{15} a_i^{lo} \, b_i^{lo}}_{\DPop{8}{8}{4}}
\end{align*}

\textbf{Step 5 --- DP8 (4$\times$4)}
\begin{align*}
\DPop{16}{16}{8} = {}
  & 2^{16}\underbrace{\sum_{i=0}^{7} a_{H,i}^{hi}\, b_i^{hi}}_{\DPop{8}{4}{4}}
  + 2^{12}\underbrace{\sum_{i=0}^{7} a_{L,i}^{hi}\, b_i^{hi}}_{\DPop{8}{4}{4}}
  + 2^{12}\underbrace{\sum_{i=0}^{7} a_{H,i}^{hi}\, b_i^{lo}}_{\DPop{8}{4}{4}}
  + 2^{8}\underbrace{\sum_{i=0}^{7} a_{L,i}^{hi}\, b_i^{lo}}_{\DPop{8}{4}{4}} \\[0.6em]
  & + 2^{8}\underbrace{\sum_{i=0}^{7} a_{H,i}^{lo}\, b_i^{hi}}_{\DPop{8}{4}{4}}
  + 2^{4}\underbrace{\sum_{i=0}^{7} a_{L,i}^{lo}\, b_i^{hi}}_{\DPop{8}{4}{4}}
  + 2^{4}\underbrace{\sum_{i=0}^{7} a_{H,i}^{lo}\, b_i^{lo}}_{\DPop{8}{4}{4}}
  + \underbrace{\sum_{i=0}^{7} a_{L,i}^{lo}\, b_i^{lo}}_{\DPop{8}{4}{4}} \\[0.6em]
  & + 2^{16}\underbrace{\sum_{i=8}^{15} a_{H,i}^{hi}\, b_i^{hi}}_{\DPop{8}{4}{4}}
  + 2^{12}\underbrace{\sum_{i=8}^{15} a_{L,i}^{hi}\, b_i^{hi}}_{\DPop{8}{4}{4}}
  + 2^{12}\underbrace{\sum_{i=8}^{15} a_{H,i}^{hi}\, b_i^{lo}}_{\DPop{8}{4}{4}}
  + 2^{8}\underbrace{\sum_{i=8}^{15} a_{L,i}^{hi}\, b_i^{lo}}_{\DPop{8}{4}{4}} \\[0.6em]
  & + 2^{8}\underbrace{\sum_{i=8}^{15} a_{H,i}^{lo}\, b_i^{hi}}_{\DPop{8}{4}{4}}
  + 2^{4}\underbrace{\sum_{i=8}^{15} a_{L,i}^{lo}\, b_i^{hi}}_{\DPop{8}{4}{4}}
  + 2^{4}\underbrace{\sum_{i=8}^{15} a_{H,i}^{lo}\, b_i^{lo}}_{\DPop{8}{4}{4}}
  + \underbrace{\sum_{i=8}^{15} a_{L,i}^{lo}\, b_i^{lo}}_{\DPop{8}{4}{4}}
\end{align*}

\textbf{Step 6 --- Add-and-square}\quad
{\small(substitute each DP8 (4$\times$4) $= \tfrac{1}{2}\big[\sum(x_i{+}b_i)^2 - \sum x_i^2 - \sum b_i^2\big]$)}
\begin{align*}
\DPop{16}{16}{8} = {}
  & 2^{16}\underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{H,i}^{hi}{+}b_i^{hi})^2 - \sum_{i=0}^{7}(a_{H,i}^{hi})^2 - \sum_{i=0}^{7}(b_i^{hi})^2\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + 2^{12}\underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{L,i}^{hi}{+}b_i^{hi})^2 - \sum_{i=0}^{7}(a_{L,i}^{hi})^2 - \sum_{i=0}^{7}(b_i^{hi})^2\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + 2^{12}\underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{H,i}^{hi}{+}b_i^{lo})^2 - \sum_{i=0}^{7}(a_{H,i}^{hi})^2 - \sum_{i=0}^{7}(b_i^{lo})^2\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + 2^{8}\underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{L,i}^{hi}{+}b_i^{lo})^2 - \sum_{i=0}^{7}(a_{L,i}^{hi})^2 - \sum_{i=0}^{7}(b_i^{lo})^2\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + 2^{8}\underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{H,i}^{lo}{+}b_i^{hi})^2 - \sum_{i=0}^{7}(a_{H,i}^{lo})^2 - \sum_{i=0}^{7}(b_i^{hi})^2\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + 2^{4}\underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{L,i}^{lo}{+}b_i^{hi})^2 - \sum_{i=0}^{7}(a_{L,i}^{lo})^2 - \sum_{i=0}^{7}(b_i^{hi})^2\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + 2^{4}\underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{H,i}^{lo}{+}b_i^{lo})^2 - \sum_{i=0}^{7}(a_{H,i}^{lo})^2 - \sum_{i=0}^{7}(b_i^{lo})^2\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + \underbrace{\tfrac{1}{2}\Big(\sum_{i=0}^{7}(a_{L,i}^{lo}{+}b_i^{lo})^2 - \sum_{i=0}^{7}(a_{L,i}^{lo})^2 - \sum_{i=0}^{7}(b_i^{lo})^2\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + 2^{16}\underbrace{\tfrac{1}{2}\Big(\sum_{i=8}^{15}(a_{H,i}^{hi}{+}b_i^{hi})^2 - \sum_{i=8}^{15}(a_{H,i}^{hi})^2 - \sum_{i=8}^{15}(b_i^{hi})^2\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + 2^{12}\underbrace{\tfrac{1}{2}\Big(\sum_{i=8}^{15}(a_{L,i}^{hi}{+}b_i^{hi})^2 - \sum_{i=8}^{15}(a_{L,i}^{hi})^2 - \sum_{i=8}^{15}(b_i^{hi})^2\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + 2^{12}\underbrace{\tfrac{1}{2}\Big(\sum_{i=8}^{15}(a_{H,i}^{hi}{+}b_i^{lo})^2 - \sum_{i=8}^{15}(a_{H,i}^{hi})^2 - \sum_{i=8}^{15}(b_i^{lo})^2\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + 2^{8}\underbrace{\tfrac{1}{2}\Big(\sum_{i=8}^{15}(a_{L,i}^{hi}{+}b_i^{lo})^2 - \sum_{i=8}^{15}(a_{L,i}^{hi})^2 - \sum_{i=8}^{15}(b_i^{lo})^2\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + 2^{8}\underbrace{\tfrac{1}{2}\Big(\sum_{i=8}^{15}(a_{H,i}^{lo}{+}b_i^{hi})^2 - \sum_{i=8}^{15}(a_{H,i}^{lo})^2 - \sum_{i=8}^{15}(b_i^{hi})^2\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + 2^{4}\underbrace{\tfrac{1}{2}\Big(\sum_{i=8}^{15}(a_{L,i}^{lo}{+}b_i^{hi})^2 - \sum_{i=8}^{15}(a_{L,i}^{lo})^2 - \sum_{i=8}^{15}(b_i^{hi})^2\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + 2^{4}\underbrace{\tfrac{1}{2}\Big(\sum_{i=8}^{15}(a_{H,i}^{lo}{+}b_i^{lo})^2 - \sum_{i=8}^{15}(a_{H,i}^{lo})^2 - \sum_{i=8}^{15}(b_i^{lo})^2\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
  & + \underbrace{\tfrac{1}{2}\Big(\sum_{i=8}^{15}(a_{L,i}^{lo}{+}b_i^{lo})^2 - \sum_{i=8}^{15}(a_{L,i}^{lo})^2 - \sum_{i=8}^{15}(b_i^{lo})^2\Big)}_{\DPop{8}{4}{4}}
\end{align*}

\textbf{Step 7 --- All components}
\[
\boxed{\;
\begin{aligned}
\mathrm{PE} ={}& 2^{16}\sum_{i=0}^{7}(a_{H,i}^{hi}{+}b_i^{hi})^2 + 2^{12}\sum_{i=0}^{7}(a_{L,i}^{hi}{+}b_i^{hi})^2 + 2^{12}\sum_{i=0}^{7}(a_{H,i}^{hi}{+}b_i^{lo})^2 + 2^{8}\sum_{i=0}^{7}(a_{L,i}^{hi}{+}b_i^{lo})^2 \\
  & + 2^{8}\sum_{i=0}^{7}(a_{H,i}^{lo}{+}b_i^{hi})^2 + 2^{4}\sum_{i=0}^{7}(a_{L,i}^{lo}{+}b_i^{hi})^2 + 2^{4}\sum_{i=0}^{7}(a_{H,i}^{lo}{+}b_i^{lo})^2 + \sum_{i=0}^{7}(a_{L,i}^{lo}{+}b_i^{lo})^2 \\
  & + 2^{16}\sum_{i=8}^{15}(a_{H,i}^{hi}{+}b_i^{hi})^2 + 2^{12}\sum_{i=8}^{15}(a_{L,i}^{hi}{+}b_i^{hi})^2 + 2^{12}\sum_{i=8}^{15}(a_{H,i}^{hi}{+}b_i^{lo})^2 + 2^{8}\sum_{i=8}^{15}(a_{L,i}^{hi}{+}b_i^{lo})^2 \\
  & + 2^{8}\sum_{i=8}^{15}(a_{H,i}^{lo}{+}b_i^{hi})^2 + 2^{4}\sum_{i=8}^{15}(a_{L,i}^{lo}{+}b_i^{hi})^2 + 2^{4}\sum_{i=8}^{15}(a_{H,i}^{lo}{+}b_i^{lo})^2 + \sum_{i=8}^{15}(a_{L,i}^{lo}{+}b_i^{lo})^2 \\[0.6em]
\alpha ={}& 2^{16}\sum_{i=0}^{7}(a_{H,i}^{hi})^2 + 2^{12}\sum_{i=0}^{7}(a_{L,i}^{hi})^2 + 2^{12}\sum_{i=0}^{7}(a_{H,i}^{hi})^2 + 2^{8}\sum_{i=0}^{7}(a_{L,i}^{hi})^2 \\
  & + 2^{8}\sum_{i=0}^{7}(a_{H,i}^{lo})^2 + 2^{4}\sum_{i=0}^{7}(a_{L,i}^{lo})^2 + 2^{4}\sum_{i=0}^{7}(a_{H,i}^{lo})^2 + \sum_{i=0}^{7}(a_{L,i}^{lo})^2 \\
  & + 2^{16}\sum_{i=8}^{15}(a_{H,i}^{hi})^2 + 2^{12}\sum_{i=8}^{15}(a_{L,i}^{hi})^2 + 2^{12}\sum_{i=8}^{15}(a_{H,i}^{hi})^2 + 2^{8}\sum_{i=8}^{15}(a_{L,i}^{hi})^2 \\
  & + 2^{8}\sum_{i=8}^{15}(a_{H,i}^{lo})^2 + 2^{4}\sum_{i=8}^{15}(a_{L,i}^{lo})^2 + 2^{4}\sum_{i=8}^{15}(a_{H,i}^{lo})^2 + \sum_{i=8}^{15}(a_{L,i}^{lo})^2 \\[0.6em]
\beta ={}& 2^{16}\sum_{i=0}^{7}(b_i^{hi})^2 + 2^{12}\sum_{i=0}^{7}(b_i^{hi})^2 + 2^{12}\sum_{i=0}^{7}(b_i^{lo})^2 + 2^{8}\sum_{i=0}^{7}(b_i^{lo})^2 \\
  & + 2^{8}\sum_{i=0}^{7}(b_i^{hi})^2 + 2^{4}\sum_{i=0}^{7}(b_i^{hi})^2 + 2^{4}\sum_{i=0}^{7}(b_i^{lo})^2 + \sum_{i=0}^{7}(b_i^{lo})^2 \\
  & + 2^{16}\sum_{i=8}^{15}(b_i^{hi})^2 + 2^{12}\sum_{i=8}^{15}(b_i^{hi})^2 + 2^{12}\sum_{i=8}^{15}(b_i^{lo})^2 + 2^{8}\sum_{i=8}^{15}(b_i^{lo})^2 \\
  & + 2^{8}\sum_{i=8}^{15}(b_i^{hi})^2 + 2^{4}\sum_{i=8}^{15}(b_i^{hi})^2 + 2^{4}\sum_{i=8}^{15}(b_i^{lo})^2 + \sum_{i=8}^{15}(b_i^{lo})^2
\end{aligned}
\;}
\]
