\begin{center}
{\LARGE \textbf{Mode 1 --- Square (centered / opt)}}\\[4pt]
{\large DP16 (8$\times$4) $\rightarrow$ DP8 (8$\times$4) $\rightarrow$ DP8 (4$\times$4) $\rightarrow$ centered add-and-square}
\end{center}

\textbf{Step 1 --- DP16 (8$\times$4)}
\[ \DPop{16}{8}{4} = \sum_{i=0}^{15} a_i \, b_i \]

\textbf{Step 2 --- DP8 (8$\times$4)}
\[ \DPop{16}{8}{4} = \underbrace{\sum_{i=0}^{7} a_i b_i}_{\DPop{8}{8}{4}} + \underbrace{\sum_{i=8}^{15} a_i b_i}_{\DPop{8}{8}{4}} \]

\textbf{Step 3 --- DP8 (4$\times$4)}
\begin{align*}
\DPop{16}{8}{4} = {}
 & 2^{4}\underbrace{\sum_{i=0}^{7} a_{H,i} b_i}_{\DPop{8}{4}{4}} + \underbrace{\sum_{i=0}^{7} a_{L,i} b_i}_{\DPop{8}{4}{4}}
 + 2^{4}\underbrace{\sum_{i=8}^{15} a_{H,i} b_i}_{\DPop{8}{4}{4}} + \underbrace{\sum_{i=8}^{15} a_{L,i} b_i}_{\DPop{8}{4}{4}}
\end{align*}

\textbf{Step 4 --- Centered add-and-square}\quad
{\small(subtract $8$ from $a_L$; each DP8 (4$\times$4) $=\tfrac12\big[\sum(x{+}b)^2-\sum x^2-\sum b^2\,(+512\text{ for }a_L)\big]$)}
\begin{align*}
\DPop{16}{8}{4} = {}
  & 2^{4}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}(a_{H,i}{+}b_i)^2 - \sum_{i=0}^{7}a_{H,i}^2 - \sum_{i=0}^{7}b_i^2\Big)}_{\DPop{8}{4}{4}} \\[0.6em]
  & + \underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big((a_{L,i}{-}8){+}b_i\big)^2 - \sum_{i=0}^{7}(a_{L,i}{-}8)^2 - \sum_{i=0}^{7}(b_i{-}8)^2 + 512\Big)}_{\DPop{8}{4}{4}} \\[0.6em]
  & + 2^{4}\underbrace{\tfrac12\Big(\sum_{i=8}^{15}(a_{H,i}{+}b_i)^2 - \sum_{i=8}^{15}a_{H,i}^2 - \sum_{i=8}^{15}b_i^2\Big)}_{\DPop{8}{4}{4}} \\[0.6em]
  & + \underbrace{\tfrac12\Big(\sum_{i=8}^{15}\big((a_{L,i}{-}8){+}b_i\big)^2 - \sum_{i=8}^{15}(a_{L,i}{-}8)^2 - \sum_{i=8}^{15}(b_i{-}8)^2 + 512\Big)}_{\DPop{8}{4}{4}}
\end{align*}

\textbf{Step 5 --- All components}
\[
\boxed{\;
\begin{aligned}
\mathrm{PE} ={}& 2^{4}\!\sum_{i=0}^{7}(a_{H,i}{+}b_i)^2 + \sum_{i=0}^{7}\!\big((a_{L,i}{-}8){+}b_i\big)^2 + 2^{4}\!\sum_{i=8}^{15}(a_{H,i}{+}b_i)^2 + \sum_{i=8}^{15}\!\big((a_{L,i}{-}8){+}b_i\big)^2 \\[0.3em]
\alpha ={}& 2^{4}\!\sum_{i=0}^{7} a_{H,i}^2 + \sum_{i=0}^{7}(a_{L,i}{-}8)^2 + 2^{4}\!\sum_{i=8}^{15} a_{H,i}^2 + \sum_{i=8}^{15}(a_{L,i}{-}8)^2 \\[0.3em]
\beta ={}& 2^{4}\!\sum_{i=0}^{7} b_i^2 + \sum_{i=0}^{7}(b_i{-}8)^2 + 2^{4}\!\sum_{i=8}^{15} b_i^2 + \sum_{i=8}^{15}(b_i{-}8)^2 \\[0.3em]
C ={}& 0 + 512 + 0 + 512 = 1024
\end{aligned}
\;}
\]
