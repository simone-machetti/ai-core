\begin{center}
{\LARGE \textbf{Mode 12 --- Square (centered / opt)}}\\[4pt]
{\large C-DP4 (C16$\times$C16) $\rightarrow$ DP8 (16$\times$16) $\rightarrow$ DP8 (16$\times$8) $\rightarrow$ DP8 (16$\times$4) $\rightarrow$ DP8 (8$\times$4) $\rightarrow$ DP8 (4$\times$4) $\rightarrow$ centered add-and-square}
\end{center}

\vspace{0.1em}


\textbf{Step 1 --- C-DP4 (C16$\times$C16)}
\[
\text{C-DP4}\,(\mathrm{C16}\!\times\!\mathrm{C16}) = D = \sum_{i=0}^{3} a_i \, b_i, \qquad
a_i = a_i^{re} + \mathrm{j}\,a_i^{im}, \quad
b_i = b_i^{re} + \mathrm{j}\,b_i^{im}
\]

\textbf{Step 2 --- DP8 (16$\times$16)}
\[
\mathrm{Re}(D) = \underbrace{\sum_{i=0}^{3} a_i^{re} b_i^{re}
  - \sum_{i=0}^{3} a_i^{im} b_i^{im}}_{\DPop{8}{16}{16}}
\]
\[
\mathrm{Im}(D) = \underbrace{\sum_{i=0}^{3} a_i^{re} b_i^{im}
  + \sum_{i=0}^{3} a_i^{im} b_i^{re}}_{\DPop{8}{16}{16}}
\]

\textbf{Step 3 --- DP8 (16$\times$8)}
\[
\mathrm{Re}(D) = \underbrace{2^{8}\sum_{i=0}^{3} a_i^{re} b_i^{re,hi}
  - 2^{8}\sum_{i=0}^{3} a_i^{im} b_i^{im,hi}}_{\DPop{8}{16}{8}}
  + \underbrace{\sum_{i=0}^{3} a_i^{re} b_i^{re,lo}
  - \sum_{i=0}^{3} a_i^{im} b_i^{im,lo}}_{\DPop{8}{16}{8}}
\]
\[
\mathrm{Im}(D) = \underbrace{2^{8}\sum_{i=0}^{3} a_i^{re} b_i^{im,hi}
  + 2^{8}\sum_{i=0}^{3} a_i^{im} b_i^{re,hi}}_{\DPop{8}{16}{8}}
  + \underbrace{\sum_{i=0}^{3} a_i^{re} b_i^{im,lo}
  + \sum_{i=0}^{3} a_i^{im} b_i^{re,lo}}_{\DPop{8}{16}{8}}
\]

\begin{align*}
