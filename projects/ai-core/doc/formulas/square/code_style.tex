\documentclass[fleqn]{article}
\usepackage{amsmath}

\setlength{\belowdisplayskip}{3pt plus 1pt minus 1pt}
\setlength{\parindent}{0pt}

\newcommand{\DPop}[3]{\mathrm{DP#1}\,(#2\!\times\!#3)}   % DP<len> (<wA> x <wB>)

\begin{document}

\begin{center}
{\LARGE \textbf{Square Formula Sheet --- Format}}\\[4pt]
{\large canonical layout for \texttt{doc/formulas/square/mode\_*.tex} \quad (example: \texttt{mode\_1.tex})}
\end{center}

\vspace{1em}

\begin{enumerate}\itemsep3pt
\item Title \texttt{Mode $N$ --- Square}; subtitle $=$ the DP granularity flow, coarsest to finest, ending in \emph{add-and-square}.
\item The top DP operation stays on the \textbf{left} of every step; refine only the \textbf{right} side, one granularity per step, down to $\DPop{8}{4}{4}$.
\item Step titles carry \textbf{no} parenthetical note --- \emph{except} the add-and-square step (keeps the substitution note).
\item Sums in \textbf{display} style (limits stacked); \textbf{underbrace} each DP-level block with \verb|\DPop{len}{wA}{wB}|.
\item Add-and-square step: \textbf{fully expanded}, top DP on the left, the $\tfrac12$ \textbf{inside} each underbrace (so the braced group is one $\DPop{8}{4}{4}$), the $2^{k}$ weight \textbf{outside}.
\item Final step \emph{All components}: \fbox{$\mathrm{PE}\ /\ \alpha\ /\ \beta$} --- $\mathrm{PE}=$ squares, $\alpha=$ A-only, $\beta=$ B-only; one term per $\DPop{8}{4}{4}$, same order in every row.
\item Keep to one page where possible: tune the \verb|\\[..]| gaps and the margin.
\end{enumerate}
