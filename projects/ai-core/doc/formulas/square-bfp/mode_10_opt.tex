\begin{center}
{\LARGE \textbf{Mode 10 --- Square-BFP (centered / opt)}}\\[4pt]
{\large C-DP8 (C8$\times$C8) $\rightarrow$ DP16 (8$\times$8) $\rightarrow$ DP16 (8$\times$4) $\rightarrow$ DP8 (8$\times$4) $\rightarrow$ block exponents $\rightarrow$ DP8 (4$\times$4) $\rightarrow$ centered add-and-square}
\end{center}

\vspace{0.1em}

\textbf{Step 1 --- C-DP8 (C8$\times$C8)}
\[
\text{C-DP8}\,(\mathrm{C8}\!\times\!\mathrm{C8}) = D = \sum_{i=0}^{7} a_i \, b_i, \qquad
a_i = a_i^{re} + \mathrm{j}\,a_i^{im}, \quad
b_i = b_i^{re} + \mathrm{j}\,b_i^{im}
\]

\textbf{Step 2 --- DP16 (8$\times$8)}
\[
\mathrm{Re}(D) = \underbrace{\sum_{i=0}^{7} a_i^{re}\, b_i^{re}
  - \sum_{i=0}^{7} a_i^{im}\, b_i^{im}}_{\DPop{16}{8}{8}}
\]
\[
\mathrm{Im}(D) = \underbrace{\sum_{i=0}^{7} a_i^{re}\, b_i^{im}
  + \sum_{i=0}^{7} a_i^{im}\, b_i^{re}}_{\DPop{16}{8}{8}}
\]

\textbf{Step 3 --- DP16 (8$\times$4)}
\[
\mathrm{Re}(D) = 2^{4}\underbrace{\sum_{i=0}^{7} a_i^{re}\, b_i^{re,hi}
  - \sum_{i=0}^{7} a_i^{im}\, b_i^{im,hi}}_{\DPop{16}{8}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i^{re}\, b_i^{re,lo}
  - \sum_{i=0}^{7} a_i^{im}\, b_i^{im,lo}}_{\DPop{16}{8}{4}}
\]
\[
\mathrm{Im}(D) = 2^{4}\underbrace{\sum_{i=0}^{7} a_i^{re}\, b_i^{im,hi}
  + \sum_{i=0}^{7} a_i^{im}\, b_i^{re,hi}}_{\DPop{16}{8}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i^{re}\, b_i^{im,lo}
  + \sum_{i=0}^{7} a_i^{im}\, b_i^{re,lo}}_{\DPop{16}{8}{4}}
\]

\textbf{Step 4 --- DP8 (8$\times$4)}
\begin{align*}
\mathrm{Re}(D) = {}
  & 2^{4}\underbrace{\sum_{i=0}^{7} a_i^{re}\, b_i^{re,hi}}_{\DPop{8}{8}{4}}
  - 2^{4}\underbrace{\sum_{i=0}^{7} a_i^{im}\, b_i^{im,hi}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i^{re}\, b_i^{re,lo}}_{\DPop{8}{8}{4}}
  - \underbrace{\sum_{i=0}^{7} a_i^{im}\, b_i^{im,lo}}_{\DPop{8}{8}{4}}
\end{align*}
\begin{align*}
\mathrm{Im}(D) = {}
  & 2^{4}\underbrace{\sum_{i=0}^{7} a_i^{re}\, b_i^{im,hi}}_{\DPop{8}{8}{4}}
  + 2^{4}\underbrace{\sum_{i=0}^{7} a_i^{im}\, b_i^{re,hi}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i^{re}\, b_i^{im,lo}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i^{im}\, b_i^{re,lo}}_{\DPop{8}{8}{4}}
\end{align*}

\textbf{Step 5 --- Block exponents}\quad
{\small(one exponent block per DP8 (8$\times$4) --- all copies of one $E$; re/im tied, negation scale-blind; parallel matmul: own $E_a,E_b$)}
\[
\big(P_j,\; E\big)\ \ \text{per}\ \DPop{8}{8}{4}\ \text{above}, \qquad E = E_a + E_b
\;\longrightarrow\;
\big(M^{re},\; E\big), \quad \big(M^{im},\; E\big)
\]
$\delta\equiv0$: $M^{re}$, $M^{im}$ are the Step-4 sums unchanged; \quad $\mathrm{Re}(\hat D)=M^{re}\cdot2^{E}$, \ $\mathrm{Im}(\hat D)=M^{im}\cdot2^{E}$.

\textbf{Step 6 --- DP8 (4$\times$4)}
\begin{align*}
M^{re} = {}
  & 2^{8}\underbrace{\sum_{i=0}^{7} a_{H,i}^{re}\, b_i^{re,hi}}_{\DPop{8}{4}{4}}
  + 2^{4}\underbrace{\sum_{i=0}^{7} a_{L,i}^{re}\, b_i^{re,hi}}_{\DPop{8}{4}{4}}
  - 2^{8}\underbrace{\sum_{i=0}^{7} a_{H,i}^{im}\, b_i^{im,hi}}_{\DPop{8}{4}{4}}
  - 2^{4}\underbrace{\sum_{i=0}^{7} a_{L,i}^{im}\, b_i^{im,hi}}_{\DPop{8}{4}{4}} \\[0.2em]
  & + 2^{4}\underbrace{\sum_{i=0}^{7} a_{H,i}^{re}\, b_i^{re,lo}}_{\DPop{8}{4}{4}}
  + \underbrace{\sum_{i=0}^{7} a_{L,i}^{re}\, b_i^{re,lo}}_{\DPop{8}{4}{4}}
  - 2^{4}\underbrace{\sum_{i=0}^{7} a_{H,i}^{im}\, b_i^{im,lo}}_{\DPop{8}{4}{4}}
  - \underbrace{\sum_{i=0}^{7} a_{L,i}^{im}\, b_i^{im,lo}}_{\DPop{8}{4}{4}}
\end{align*}
\begin{align*}
M^{im} = {}
  & 2^{8}\underbrace{\sum_{i=0}^{7} a_{H,i}^{re}\, b_i^{im,hi}}_{\DPop{8}{4}{4}}
  + 2^{4}\underbrace{\sum_{i=0}^{7} a_{L,i}^{re}\, b_i^{im,hi}}_{\DPop{8}{4}{4}}
  + 2^{8}\underbrace{\sum_{i=0}^{7} a_{H,i}^{im}\, b_i^{re,hi}}_{\DPop{8}{4}{4}}
  + 2^{4}\underbrace{\sum_{i=0}^{7} a_{L,i}^{im}\, b_i^{re,hi}}_{\DPop{8}{4}{4}} \\[0.2em]
  & + 2^{4}\underbrace{\sum_{i=0}^{7} a_{H,i}^{re}\, b_i^{im,lo}}_{\DPop{8}{4}{4}}
  + \underbrace{\sum_{i=0}^{7} a_{L,i}^{re}\, b_i^{im,lo}}_{\DPop{8}{4}{4}}
  + 2^{4}\underbrace{\sum_{i=0}^{7} a_{H,i}^{im}\, b_i^{re,lo}}_{\DPop{8}{4}{4}}
  + \underbrace{\sum_{i=0}^{7} a_{L,i}^{im}\, b_i^{re,lo}}_{\DPop{8}{4}{4}}
\end{align*}

\textbf{Step 7 --- Centered add-and-square}\quad
{\small(subtract $8$ from unsigned nibbles; per DP8 (4$\times$4) add $c=0/512/2048$ for 0/1/2 unsigned nibbles. The $-\Sigma\,a^{im}b^{im}$ group keeps the add-square but carries a global $-$.)}
\begin{align*}
M^{re} = {}
  & 2^{8}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}(a_{H,i}^{re}{+}b_i^{re,hi})^2 - \sum_{i=0}^{7}(a_{H,i}^{re})^2 - \sum_{i=0}^{7}(b_i^{re,hi})^2\Big)}_{\DPop{8}{4}{4}} \\[0.4em]
  & + 2^{4}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big((a_{L,i}^{re}{-}8){+}b_i^{re,hi}\big)^2 - \sum_{i=0}^{7}(a_{L,i}^{re}{-}8)^2 - \sum_{i=0}^{7}(b_i^{re,hi}{-}8)^2 + 512\Big)}_{\DPop{8}{4}{4}} \\[0.4em]
  & - 2^{8}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}(a_{H,i}^{im}{+}b_i^{im,hi})^2 - \sum_{i=0}^{7}(a_{H,i}^{im})^2 - \sum_{i=0}^{7}(b_i^{im,hi})^2\Big)}_{\DPop{8}{4}{4}} \\[0.4em]
  & - 2^{4}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big((a_{L,i}^{im}{-}8){+}b_i^{im,hi}\big)^2 - \sum_{i=0}^{7}(a_{L,i}^{im}{-}8)^2 - \sum_{i=0}^{7}(b_i^{im,hi}{-}8)^2 + 512\Big)}_{\DPop{8}{4}{4}} \\[0.4em]
  & + 2^{4}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big(a_{H,i}^{re}{+}(b_i^{re,lo}{-}8)\big)^2 - \sum_{i=0}^{7}(a_{H,i}^{re}{-}8)^2 - \sum_{i=0}^{7}(b_i^{re,lo}{-}8)^2 + 512\Big)}_{\DPop{8}{4}{4}} \\[0.4em]
  & + \underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big((a_{L,i}^{re}{-}8){+}(b_i^{re,lo}{-}8)\big)^2 - \sum_{i=0}^{7}(a_{L,i}^{re}{-}16)^2 - \sum_{i=0}^{7}(b_i^{re,lo}{-}16)^2 + 2048\Big)}_{\DPop{8}{4}{4}} \\[0.4em]
  & - 2^{4}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big(a_{H,i}^{im}{+}(b_i^{im,lo}{-}8)\big)^2 - \sum_{i=0}^{7}(a_{H,i}^{im}{-}8)^2 - \sum_{i=0}^{7}(b_i^{im,lo}{-}8)^2 + 512\Big)}_{\DPop{8}{4}{4}} \\[0.4em]
  & - \underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big((a_{L,i}^{im}{-}8){+}(b_i^{im,lo}{-}8)\big)^2 - \sum_{i=0}^{7}(a_{L,i}^{im}{-}16)^2 - \sum_{i=0}^{7}(b_i^{im,lo}{-}16)^2 + 2048\Big)}_{\DPop{8}{4}{4}}
\end{align*}
\begin{align*}
M^{im} = {}
  & 2^{8}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}(a_{H,i}^{re}{+}b_i^{im,hi})^2 - \sum_{i=0}^{7}(a_{H,i}^{re})^2 - \sum_{i=0}^{7}(b_i^{im,hi})^2\Big)}_{\DPop{8}{4}{4}} \\[0.4em]
  & + 2^{4}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big((a_{L,i}^{re}{-}8){+}b_i^{im,hi}\big)^2 - \sum_{i=0}^{7}(a_{L,i}^{re}{-}8)^2 - \sum_{i=0}^{7}(b_i^{im,hi}{-}8)^2 + 512\Big)}_{\DPop{8}{4}{4}} \\[0.4em]
  & + 2^{8}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}(a_{H,i}^{im}{+}b_i^{re,hi})^2 - \sum_{i=0}^{7}(a_{H,i}^{im})^2 - \sum_{i=0}^{7}(b_i^{re,hi})^2\Big)}_{\DPop{8}{4}{4}} \\[0.4em]
  & + 2^{4}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big((a_{L,i}^{im}{-}8){+}b_i^{re,hi}\big)^2 - \sum_{i=0}^{7}(a_{L,i}^{im}{-}8)^2 - \sum_{i=0}^{7}(b_i^{re,hi}{-}8)^2 + 512\Big)}_{\DPop{8}{4}{4}} \\[0.4em]
  & + 2^{4}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big(a_{H,i}^{re}{+}(b_i^{im,lo}{-}8)\big)^2 - \sum_{i=0}^{7}(a_{H,i}^{re}{-}8)^2 - \sum_{i=0}^{7}(b_i^{im,lo}{-}8)^2 + 512\Big)}_{\DPop{8}{4}{4}} \\[0.4em]
  & + \underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big((a_{L,i}^{re}{-}8){+}(b_i^{im,lo}{-}8)\big)^2 - \sum_{i=0}^{7}(a_{L,i}^{re}{-}16)^2 - \sum_{i=0}^{7}(b_i^{im,lo}{-}16)^2 + 2048\Big)}_{\DPop{8}{4}{4}} \\[0.4em]
  & + 2^{4}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big(a_{H,i}^{im}{+}(b_i^{re,lo}{-}8)\big)^2 - \sum_{i=0}^{7}(a_{H,i}^{im}{-}8)^2 - \sum_{i=0}^{7}(b_i^{re,lo}{-}8)^2 + 512\Big)}_{\DPop{8}{4}{4}} \\[0.4em]
  & + \underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big((a_{L,i}^{im}{-}8){+}(b_i^{re,lo}{-}8)\big)^2 - \sum_{i=0}^{7}(a_{L,i}^{im}{-}16)^2 - \sum_{i=0}^{7}(b_i^{re,lo}{-}16)^2 + 2048\Big)}_{\DPop{8}{4}{4}}
\end{align*}

\textbf{Step 8 --- All components}

\textbf{Re} ($M^{re}$):\quad{\small(the $a^{im}b^{im}$ blocks carry a global $-$ in $\mathrm{PE}$, $\alpha$, $\beta$ and $C$)}
\[
\boxed{\;
\begin{aligned}
\mathrm{PE} ={}& 2^{8}\!\sum_{i=0}^{7}(a_{H,i}^{re}{+}b_i^{re,hi})^2 + 2^{4}\!\sum_{i=0}^{7}\big((a_{L,i}^{re}{-}8){+}b_i^{re,hi}\big)^2 - 2^{8}\!\sum_{i=0}^{7}(a_{H,i}^{im}{+}b_i^{im,hi})^2 - 2^{4}\!\sum_{i=0}^{7}\big((a_{L,i}^{im}{-}8){+}b_i^{im,hi}\big)^2 \\
  & + 2^{4}\!\sum_{i=0}^{7}\big(a_{H,i}^{re}{+}(b_i^{re,lo}{-}8)\big)^2 + \sum_{i=0}^{7}\big((a_{L,i}^{re}{-}8){+}(b_i^{re,lo}{-}8)\big)^2 - 2^{4}\!\sum_{i=0}^{7}\big(a_{H,i}^{im}{+}(b_i^{im,lo}{-}8)\big)^2 - \sum_{i=0}^{7}\big((a_{L,i}^{im}{-}8){+}(b_i^{im,lo}{-}8)\big)^2 \\[0.3em]
\alpha ={}& 2^{8}\!\sum_{i=0}^{7}(a_{H,i}^{re})^2 + 2^{4}\!\sum_{i=0}^{7}(a_{L,i}^{re}{-}8)^2 - 2^{8}\!\sum_{i=0}^{7}(a_{H,i}^{im})^2 - 2^{4}\!\sum_{i=0}^{7}(a_{L,i}^{im}{-}8)^2 \\
  & + 2^{4}\!\sum_{i=0}^{7}(a_{H,i}^{re}{-}8)^2 + \sum_{i=0}^{7}(a_{L,i}^{re}{-}16)^2 - 2^{4}\!\sum_{i=0}^{7}(a_{H,i}^{im}{-}8)^2 - \sum_{i=0}^{7}(a_{L,i}^{im}{-}16)^2 \\[0.3em]
\beta ={}& 2^{8}\!\sum_{i=0}^{7}(b_i^{re,hi})^2 + 2^{4}\!\sum_{i=0}^{7}(b_i^{re,hi}{-}8)^2 - 2^{8}\!\sum_{i=0}^{7}(b_i^{im,hi})^2 - 2^{4}\!\sum_{i=0}^{7}(b_i^{im,hi}{-}8)^2 \\
  & + 2^{4}\!\sum_{i=0}^{7}(b_i^{re,lo}{-}8)^2 + \sum_{i=0}^{7}(b_i^{re,lo}{-}16)^2 - 2^{4}\!\sum_{i=0}^{7}(b_i^{im,lo}{-}8)^2 - \sum_{i=0}^{7}(b_i^{im,lo}{-}16)^2 \\[0.3em]
C ={}& 0 + 8192 + 0 - 8192 + 8192 + 2048 - 8192 - 2048 = 0 \\[0.3em]
M^{re} ={}& \tfrac12\big(\mathrm{PE} - \alpha - \beta + C\big), \qquad E = E_a + E_b
\end{aligned}
\;}
\]

\textbf{Im} ($M^{im}$):
\[
\boxed{\;
\begin{aligned}
\mathrm{PE} ={}& 2^{8}\!\sum_{i=0}^{7}(a_{H,i}^{re}{+}b_i^{im,hi})^2 + 2^{4}\!\sum_{i=0}^{7}\big((a_{L,i}^{re}{-}8){+}b_i^{im,hi}\big)^2 + 2^{8}\!\sum_{i=0}^{7}(a_{H,i}^{im}{+}b_i^{re,hi})^2 + 2^{4}\!\sum_{i=0}^{7}\big((a_{L,i}^{im}{-}8){+}b_i^{re,hi}\big)^2 \\
  & + 2^{4}\!\sum_{i=0}^{7}\big(a_{H,i}^{re}{+}(b_i^{im,lo}{-}8)\big)^2 + \sum_{i=0}^{7}\big((a_{L,i}^{re}{-}8){+}(b_i^{im,lo}{-}8)\big)^2 + 2^{4}\!\sum_{i=0}^{7}\big(a_{H,i}^{im}{+}(b_i^{re,lo}{-}8)\big)^2 + \sum_{i=0}^{7}\big((a_{L,i}^{im}{-}8){+}(b_i^{re,lo}{-}8)\big)^2 \\[0.3em]
\alpha ={}& 2^{8}\!\sum_{i=0}^{7}(a_{H,i}^{re})^2 + 2^{4}\!\sum_{i=0}^{7}(a_{L,i}^{re}{-}8)^2 + 2^{8}\!\sum_{i=0}^{7}(a_{H,i}^{im})^2 + 2^{4}\!\sum_{i=0}^{7}(a_{L,i}^{im}{-}8)^2 \\
  & + 2^{4}\!\sum_{i=0}^{7}(a_{H,i}^{re}{-}8)^2 + \sum_{i=0}^{7}(a_{L,i}^{re}{-}16)^2 + 2^{4}\!\sum_{i=0}^{7}(a_{H,i}^{im}{-}8)^2 + \sum_{i=0}^{7}(a_{L,i}^{im}{-}16)^2 \\[0.3em]
\beta ={}& 2^{8}\!\sum_{i=0}^{7}(b_i^{im,hi})^2 + 2^{4}\!\sum_{i=0}^{7}(b_i^{im,hi}{-}8)^2 + 2^{8}\!\sum_{i=0}^{7}(b_i^{re,hi})^2 + 2^{4}\!\sum_{i=0}^{7}(b_i^{re,hi}{-}8)^2 \\
  & + 2^{4}\!\sum_{i=0}^{7}(b_i^{im,lo}{-}8)^2 + \sum_{i=0}^{7}(b_i^{im,lo}{-}16)^2 + 2^{4}\!\sum_{i=0}^{7}(b_i^{re,lo}{-}8)^2 + \sum_{i=0}^{7}(b_i^{re,lo}{-}16)^2 \\[0.3em]
C ={}& 0 + 8192 + 0 + 8192 + 8192 + 2048 + 8192 + 2048 = 36864 \\[0.3em]
M^{im} ={}& \tfrac12\big(\mathrm{PE} - \alpha - \beta + C\big), \qquad E = E_a + E_b
\end{aligned}
\;}
\]
