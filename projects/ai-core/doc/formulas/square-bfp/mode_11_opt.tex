\begin{center}
{\LARGE \textbf{Mode 11 --- Square-BFP (centered / opt)}}\\[4pt]
{\large C-DP16 (C8$\times$C8) $\rightarrow$ DP32 (8$\times$8) $\rightarrow$ DP16 (8$\times$8) $\rightarrow$ DP16 (8$\times$4) $\rightarrow$ DP8 (8$\times$4) $\rightarrow$ block exponents $\rightarrow$ DP8 (4$\times$4) $\rightarrow$ centered add-and-square}
\end{center}

\vspace{0.1em}

\textbf{Step 1 --- C-DP16 (C8$\times$C8)}
\[
\text{C-DP16}\,(\mathrm{C8}\!\times\!\mathrm{C8}) = D = \sum_{i=0}^{15} a_i \, b_i, \qquad
a_i = a_i^{re} + \mathrm{j}\,a_i^{im}, \quad
b_i = b_i^{re} + \mathrm{j}\,b_i^{im}
\]

\textbf{Step 2 --- DP32 (8$\times$8)}
\[
\mathrm{Re}(D) = \underbrace{\sum_{i=0}^{15} a_i^{re} b_i^{re}
  - \sum_{i=0}^{15} a_i^{im} b_i^{im}}_{\DPop{32}{8}{8}}
\]
\[
\mathrm{Im}(D) = \underbrace{\sum_{i=0}^{15} a_i^{re} b_i^{im}
  + \sum_{i=0}^{15} a_i^{im} b_i^{re}}_{\DPop{32}{8}{8}}
\]

\textbf{Step 3 --- DP16 (8$\times$8)}
\[
\mathrm{Re}(D) = \underbrace{\sum_{i=0}^{15} a_i^{re} b_i^{re}}_{\DPop{16}{8}{8}}
  \; - \; \underbrace{\sum_{i=0}^{15} a_i^{im} b_i^{im}}_{\DPop{16}{8}{8}}
\]
\[
\mathrm{Im}(D) = \underbrace{\sum_{i=0}^{15} a_i^{re} b_i^{im}}_{\DPop{16}{8}{8}}
  \; + \; \underbrace{\sum_{i=0}^{15} a_i^{im} b_i^{re}}_{\DPop{16}{8}{8}}
\]

\textbf{Step 4 --- DP16 (8$\times$4)}
\begin{align*}
\mathrm{Re}(D) = {}
  & 2^{4} \underbrace{\sum_{i=0}^{15} a_i^{re} b_i^{re,hi}}_{\DPop{16}{8}{4}}
  + \underbrace{\sum_{i=0}^{15} a_i^{re} b_i^{re,lo}}_{\DPop{16}{8}{4}}
  - 2^{4} \underbrace{\sum_{i=0}^{15} a_i^{im} b_i^{im,hi}}_{\DPop{16}{8}{4}}
  - \underbrace{\sum_{i=0}^{15} a_i^{im} b_i^{im,lo}}_{\DPop{16}{8}{4}}
\end{align*}
\begin{align*}
\mathrm{Im}(D) = {}
  & 2^{4} \underbrace{\sum_{i=0}^{15} a_i^{re} b_i^{im,hi}}_{\DPop{16}{8}{4}}
  + \underbrace{\sum_{i=0}^{15} a_i^{re} b_i^{im,lo}}_{\DPop{16}{8}{4}}
  + 2^{4} \underbrace{\sum_{i=0}^{15} a_i^{im} b_i^{re,hi}}_{\DPop{16}{8}{4}}
  + \underbrace{\sum_{i=0}^{15} a_i^{im} b_i^{re,lo}}_{\DPop{16}{8}{4}}
\end{align*}

\textbf{Step 5 --- DP8 (8$\times$4)}
\begin{align*}
\mathrm{Re}(D) = {}
  & 2^{4} \underbrace{\sum_{i=0}^{7} a_i^{re} b_i^{re,hi}}_{\DPop{8}{8}{4}}
  + 2^{4} \underbrace{\sum_{i=8}^{15} a_i^{re} b_i^{re,hi}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i^{re} b_i^{re,lo}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=8}^{15} a_i^{re} b_i^{re,lo}}_{\DPop{8}{8}{4}} \\[1em]
  & - 2^{4} \underbrace{\sum_{i=0}^{7} a_i^{im} b_i^{im,hi}}_{\DPop{8}{8}{4}}
  - 2^{4} \underbrace{\sum_{i=8}^{15} a_i^{im} b_i^{im,hi}}_{\DPop{8}{8}{4}}
  - \underbrace{\sum_{i=0}^{7} a_i^{im} b_i^{im,lo}}_{\DPop{8}{8}{4}}
  - \underbrace{\sum_{i=8}^{15} a_i^{im} b_i^{im,lo}}_{\DPop{8}{8}{4}}
\end{align*}
\begin{align*}
\mathrm{Im}(D) = {}
  & 2^{4} \underbrace{\sum_{i=0}^{7} a_i^{re} b_i^{im,hi}}_{\DPop{8}{8}{4}}
  + 2^{4} \underbrace{\sum_{i=8}^{15} a_i^{re} b_i^{im,hi}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i^{re} b_i^{im,lo}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=8}^{15} a_i^{re} b_i^{im,lo}}_{\DPop{8}{8}{4}} \\[1em]
  & + 2^{4} \underbrace{\sum_{i=0}^{7} a_i^{im} b_i^{re,hi}}_{\DPop{8}{8}{4}}
  + 2^{4} \underbrace{\sum_{i=8}^{15} a_i^{im} b_i^{re,hi}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i^{im} b_i^{re,lo}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=8}^{15} a_i^{im} b_i^{re,lo}}_{\DPop{8}{8}{4}}
\end{align*}

\textbf{Step 6 --- Block exponents}\quad
{\small(one exponent block per DP8 (8$\times$4); $\mathrm{Re}$/$\mathrm{Im}$ share $E_k$)}
\begin{align*}
M^{re} &= 2^{-\delta_0}\Big(2^{4}\!\sum_{i=0}^{7} a_i^{re} b_i^{re,hi} + \sum_{i=0}^{7} a_i^{re} b_i^{re,lo} - 2^{4}\!\sum_{i=0}^{7} a_i^{im} b_i^{im,hi} - \sum_{i=0}^{7} a_i^{im} b_i^{im,lo}\Big) + 2^{-\delta_1}\Big(\cdots\ i\!\in\![8,15]\ \cdots\Big) \\[0.15em]
M^{im} &= 2^{-\delta_0}\Big(2^{4}\!\sum_{i=0}^{7} a_i^{re} b_i^{im,hi} + \sum_{i=0}^{7} a_i^{re} b_i^{im,lo} + 2^{4}\!\sum_{i=0}^{7} a_i^{im} b_i^{re,hi} + \sum_{i=0}^{7} a_i^{im} b_i^{re,lo}\Big) + 2^{-\delta_1}\Big(\cdots\ i\!\in\![8,15]\ \cdots\Big)
\end{align*}
$E_k=E_{a,k}+E_{b,k}$; \quad $E=\max(E_0,E_1)$, \ $\delta_k=E-E_k$; \quad truncating shifts, on block sums only; \quad $\mathrm{Re}(\hat D)=M^{re}\cdot2^{E}$, \ $\mathrm{Im}(\hat D)=M^{im}\cdot2^{E}$.
