\begin{center}
{\LARGE \textbf{Mode 9 --- Square-BFP (centered / opt)}}\\[4pt]
{\large DP8 (16$\times$16) $\rightarrow$ DP8 (16$\times$8) $\rightarrow$ DP8 (16$\times$4) $\rightarrow$ DP8 (8$\times$4) $\rightarrow$ block exponents $\rightarrow$ DP8 (4$\times$4) $\rightarrow$ centered add-and-square}
\end{center}

\vspace{0.1em}

\textbf{Step 1 --- DP8 (16$\times$16)}
\[
\DPop{8}{16}{16} = \sum_{i=0}^{7} a_i \, b_i
\]

\textbf{Step 2 --- DP8 (16$\times$8)}
\[
\DPop{8}{16}{16}
  = 2^{8} \underbrace{\sum_{i=0}^{7} a_i \, b_i^{hi}}_{\DPop{8}{16}{8}}
  \; + \; \underbrace{\sum_{i=0}^{7} a_i \, b_i^{lo}}_{\DPop{8}{16}{8}}
\]

\textbf{Step 3 --- DP8 (16$\times$4)}
\[
\DPop{8}{16}{16}
  = 2^{12} \underbrace{\sum_{i=0}^{7} a_i \, b_i^{hh}}_{\DPop{8}{16}{4}}
  + 2^{8} \underbrace{\sum_{i=0}^{7} a_i \, b_i^{hl}}_{\DPop{8}{16}{4}}
  + 2^{4} \underbrace{\sum_{i=0}^{7} a_i \, b_i^{lh}}_{\DPop{8}{16}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i \, b_i^{ll}}_{\DPop{8}{16}{4}}
\]

\textbf{Step 4 --- DP8 (8$\times$4)}
\begin{align*}
\DPop{8}{16}{16} = {}
  & 2^{20} \underbrace{\sum_{i=0}^{7} a_i^{hi} b_i^{hh}}_{\DPop{8}{8}{4}}
  + 2^{16} \underbrace{\sum_{i=0}^{7} a_i^{hi} b_i^{hl}}_{\DPop{8}{8}{4}}
  + 2^{12} \underbrace{\sum_{i=0}^{7} a_i^{lo} b_i^{hh}}_{\DPop{8}{8}{4}}
  + 2^{12} \underbrace{\sum_{i=0}^{7} a_i^{hi} b_i^{lh}}_{\DPop{8}{8}{4}} \\[1em]
  & + 2^{8} \underbrace{\sum_{i=0}^{7} a_i^{lo} b_i^{hl}}_{\DPop{8}{8}{4}}
  + 2^{8} \underbrace{\sum_{i=0}^{7} a_i^{hi} b_i^{ll}}_{\DPop{8}{8}{4}}
  + 2^{4} \underbrace{\sum_{i=0}^{7} a_i^{lo} b_i^{lh}}_{\DPop{8}{8}{4}}
  + \underbrace{\sum_{i=0}^{7} a_i^{lo} b_i^{ll}}_{\DPop{8}{8}{4}}
\end{align*}

\textbf{Step 5 --- Block exponents}\quad
{\small(one exponent block per DP8 (8$\times$4) --- all copies of one $E$; parallel matmul: own $E_a,E_b$)}
\[
\big(P_j,\; E\big)\ \ \text{per}\ \DPop{8}{8}{4}\ \text{above}, \qquad E = E_a + E_b
\;\longrightarrow\;
\big(M,\; E\big)
\]
$\delta\equiv0$: $M$ is the Step-4 sum unchanged; \quad $\DPop{8}{16}{16}=M\cdot2^{E}$.

\textbf{Step 6 --- DP8 (4$\times$4)}
\begin{align*}
M = {}
  & 2^{24}\underbrace{\sum_{i=0}^{7} a_{H,i}^{hi} b_i^{hh}}_{\DPop{8}{4}{4}}
  + 2^{20}\underbrace{\sum_{i=0}^{7} a_{L,i}^{hi} b_i^{hh}}_{\DPop{8}{4}{4}}
  + 2^{20}\underbrace{\sum_{i=0}^{7} a_{H,i}^{hi} b_i^{hl}}_{\DPop{8}{4}{4}}
  + 2^{16}\underbrace{\sum_{i=0}^{7} a_{L,i}^{hi} b_i^{hl}}_{\DPop{8}{4}{4}} \\[0.6em]
  & + 2^{16}\underbrace{\sum_{i=0}^{7} a_{H,i}^{lo} b_i^{hh}}_{\DPop{8}{4}{4}}
  + 2^{12}\underbrace{\sum_{i=0}^{7} a_{L,i}^{lo} b_i^{hh}}_{\DPop{8}{4}{4}}
  + 2^{16}\underbrace{\sum_{i=0}^{7} a_{H,i}^{hi} b_i^{lh}}_{\DPop{8}{4}{4}}
  + 2^{12}\underbrace{\sum_{i=0}^{7} a_{L,i}^{hi} b_i^{lh}}_{\DPop{8}{4}{4}} \\[0.6em]
  & + 2^{12}\underbrace{\sum_{i=0}^{7} a_{H,i}^{lo} b_i^{hl}}_{\DPop{8}{4}{4}}
  + 2^{8}\underbrace{\sum_{i=0}^{7} a_{L,i}^{lo} b_i^{hl}}_{\DPop{8}{4}{4}}
  + 2^{12}\underbrace{\sum_{i=0}^{7} a_{H,i}^{hi} b_i^{ll}}_{\DPop{8}{4}{4}}
  + 2^{8}\underbrace{\sum_{i=0}^{7} a_{L,i}^{hi} b_i^{ll}}_{\DPop{8}{4}{4}} \\[0.6em]
  & + 2^{8}\underbrace{\sum_{i=0}^{7} a_{H,i}^{lo} b_i^{lh}}_{\DPop{8}{4}{4}}
  + 2^{4}\underbrace{\sum_{i=0}^{7} a_{L,i}^{lo} b_i^{lh}}_{\DPop{8}{4}{4}}
  + 2^{4}\underbrace{\sum_{i=0}^{7} a_{H,i}^{lo} b_i^{ll}}_{\DPop{8}{4}{4}}
  + \underbrace{\sum_{i=0}^{7} a_{L,i}^{lo} b_i^{ll}}_{\DPop{8}{4}{4}}
\end{align*}

  & + 2^{12}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big((a_{L,i}^{hi}{-}8){+}(b_i^{lh}{-}8)\big)^2 - \sum_{i=0}^{7}(a_{L,i}^{hi}{-}16)^2 - \sum_{i=0}^{7}(b_i^{lh}{-}16)^2 + 2048\Big)}_{\DPop{8}{4}{4}} \\[0.5em]

  & + 2^{8}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big((a_{L,i}^{lo}{-}8){+}(b_i^{hl}{-}8)\big)^2 - \sum_{i=0}^{7}(a_{L,i}^{lo}{-}16)^2 - \sum_{i=0}^{7}(b_i^{hl}{-}16)^2 + 2048\Big)}_{\DPop{8}{4}{4}} \\[0.5em]

  & + 2^{8}\underbrace{\tfrac12\Big(\sum_{i=0}^{7}\big((a_{H,i}^{lo}{-}8){+}(b_i^{lh}{-}8)\big)^2 - \sum_{i=0}^{7}(a_{H,i}^{lo}{-}16)^2 - \sum_{i=0}^{7}(b_i^{lh}{-}16)^2 + 2048\Big)}_{\DPop{8}{4}{4}} \\[0.5em]
